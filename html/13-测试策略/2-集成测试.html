集成测试 ​

集成测试验证 EF Core 应用与真实数据库的交互,确保数据访问层在实际环境中正常工作。本章将深入探讨如何构建高效的集成测试框架,包括测试数据库管理、事务回滚、数据隔离和CI/CD集成。

1. 集成测试vs单元测试 ​

1.1 对比分析 ​

特性单元测试集成测试
范围单个类/方法多个组件协作
数据库Mock/InMemory真实数据库
速度毫秒级秒级
目的验证逻辑正确性验证组件集成
数量大量(70%)适量(20%)
维护成本低中

1.2 何时使用集成测试? ​

✅ 适合集成测试的场景:

  1. EF Core配置验证

    • 实体映射是否正确
    • 关系配置是否生效
    • 约束是否起作用

  2. LINQ查询翻译

    • 复杂查询能否翻译为SQL
    • 性能是否符合预期
    • 生成的SQL是否正确

  3. 迁移验证

    • 迁移是否能应用
    • 数据是否保留
    • 回滚是否工作

  4. 事务行为

    • 并发控制
    • 锁机制
    • 隔离级别

❌ 不需要集成测试:

  • 纯业务逻辑(用单元测试)
  • 字符串处理
  • 数学计算
  • 数据转换

4. 事务回滚模式 ​

4.1 自动回滚模式 ​

每个测试在事务中执行,测试后回滚:

csharp
public abstract class TransactionalTestBase : IAsyncLifetime
{
    protected AppDbContext Context { get; private set; }
    private IDbContextTransaction _transaction;
    
    private readonly string _connectionString;
    
    protected TransactionalTestBase(string connectionString)
    {
        _connectionString = connectionString;
    }
    
    public async Task InitializeAsync()
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlServer(_connectionString)
            .Options;
        
        Context = new AppDbContext(options);
        
        // 开启事务
        _transaction = await Context.Database.BeginTransactionAsync();
    }
    
    public async Task DisposeAsync()
    {
        // 回滚事务 - 清除所有更改
        if (_transaction != null)
        {
            await _transaction.RollbackAsync();
            await _transaction.DisposeAsync();
        }
        
        await Context.DisposeAsync();
    }
}

// 使用
[Collection("DatabaseCollection")]
public class OrderServiceTests : TransactionalTestBase
{
    public OrderServiceTests(DatabaseContainerFixture fixture) 
        : base(fixture.ConnectionString)
    {
    }
    
    [Fact]
    public async Task CreateOrderAsync_PersistsCorrectly()
    {
        // Arrange
        var service = new OrderService(Context);
        
        // Act
        var order = await service.CreateOrderAsync(new CreateOrderRequest
        {
            CustomerId = 1,
            Items = new List<OrderItemRequest>
            {
                new() { ProductId = 1, Quantity = 2 }
            }
        });
        
        // Assert
        Assert.NotNull(order);
        Assert.True(order.Id > 0);
        
        // 测试结束后自动回滚,数据库保持干净
    }
    
    [Fact]
    public async Task AnotherTest_IsolationGuaranteed()
    {
        // 这个测试看不到上一个测试的数据
        var orders = await Context.Orders.ToListAsync();
        Assert.Empty(orders);  // ✅ 空,因为上一个测试已回滚
    }
}

4.2 Savepoint模式 ​

部分回滚到保存点:

csharp
[Fact]
public async Task TestWithPartialRollback()
{
    using var transaction = await Context.Database.BeginTransactionAsync();
    
    try
    {
        // 步骤1: 创建分类
        var category = new Category { Name = "Electronics" };
        Context.Categories.Add(category);
        await Context.SaveChangesAsync();
        
        // 创建保存点
        await transaction.CreateSavepointAsync("AfterCategory");
        
        // 步骤2: 创建产品(可能失败)
        try
        {
            var product = new Product 
            { 
                Name = "Laptop", 
                CategoryId = category.Id,
                Price = -100  // 无效价格,触发验证失败
            };
            
            Context.Products.Add(product);
            await Context.SaveChangesAsync();
        }
        catch
        {
            // 回滚到保存点,保留分类
            await transaction.RollbackToSavepointAsync("AfterCategory");
        }
        
        // Assert: 分类仍存在,产品不存在
        Assert.NotNull(await Context.Categories.FindAsync(category.Id));
        Assert.Empty(Context.Products);
        
        await transaction.CommitAsync();
    }
    catch
    {
        await transaction.RollbackAsync();
        throw;
    }
}

public class AuthenticatedApiTests

public static class TestSeedData
{
    public static async Task SeedTestDataAsync(AppDbContext context)
    {
        // 清空现有数据
        context.Orders.RemoveRange(context.Orders);
        context.Products.RemoveRange(context.Products);
        context.Categories.RemoveRange(context.Categories);
        await context.SaveChangesAsync();
        
        // 插入测试数据
        var categories = new List<Category>
        {
            new() { Id = 1, Name = "Electronics" },
            new() { Id = 2, Name = "Books" }

            new() { Id = 2, Name = "Phone", Price = 699.99m, CategoryId = 1, Stock = 100 },
            new() { Id = 3, Name = "C# Book", Price = 49.99m, CategoryId = 2, Stock = 30 }

        context.Products.AddRange(products);
        await context.SaveChangesAsync();
    }
}

public class PerformanceTests : IClassFixture<DatabaseContainerFixture>
{
    private readonly DatabaseContainerFixture _fixture;
    
    public PerformanceTests(DatabaseContainerFixture fixture)
    {
        _fixture = fixture;
    }
    
    [Fact]
    public async Task BulkInsert_PerformanceTest()
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlServer(_fixture.ConnectionString)
            .Options;
        
        using var context = new AppDbContext(options);
        
        // 预热
        await SeedSmallDatasetAsync(context);
        
        // 测试: 插入10000条记录
        var stopwatch = Stopwatch.StartNew();
        
        var products = Enumerable.Range(1, 10000)
            .Select(i => new Product 
            { 
                Name = $"Product {i}", 
                Price = i * 10,
                Stock = 100 
            })
            .ToList();
        
        context.Products.AddRange(products);
        await context.SaveChangesAsync();
        
        stopwatch.Stop();
        
        // Assert: 应该在合理时间内完成
        Assert.True(stopwatch.ElapsedMilliseconds < 5000, 
            $"Bulk insert took {stopwatch.ElapsedMilliseconds}ms");
        
        Console.WriteLine($"Inserted 10000 records in {stopwatch.ElapsedMilliseconds}ms");
    }
    
    [Fact]
    public async Task ComplexQuery_PerformanceTest()
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlServer(_fixture.ConnectionString)
            .Options;
        
        using var context = new AppDbContext(options);
        
        // 准备大数据
        await SeedLargeDatasetAsync(context);
        
        // 测试复杂查询
        var stopwatch = Stopwatch.StartNew();
        
        var results = await context.Orders
            .Where(o => o.OrderDate >= DateTime.UtcNow.AddDays(-30))
            .Include(o => o.Items)
            .ThenInclude(i => i.Product)
            .OrderByDescending(o => o.TotalAmount)
            .Take(100)
            .ToListAsync();
        
        stopwatch.Stop();
        
        Assert.True(stopwatch.ElapsedMilliseconds < 1000,
            $"Complex query took {stopwatch.ElapsedMilliseconds}ms");
        
        Console.WriteLine($"Query executed in {stopwatch.ElapsedMilliseconds}ms");
    }
}

[MemoryDiagnoser]
public class EfCoreBenchmarks

    public void Setup()
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlServer("Server=localhost;Database=Benchmarks;")
            .Options;
        
        _context = new AppDbContext(options);
        _context.Database.EnsureCreated();
        
        _products = GenerateProducts(1000);
    }
    
    [Benchmark]
    public async Task AddEntities()
    {
        _context.Products.AddRange(_products);
        await _context.SaveChangesAsync();
    }
    
    [Benchmark]
    public async Task QueryEntities()
    {
        var results = await _context.Products
            .Where(p => p.Price > 50)
            .ToListAsync();
    }
    
    [GlobalCleanup]
    public void Cleanup()
    {
        _context.Database.EnsureDeleted();
        _context.Dispose();
    }
}

// 运行基准测试
public class Program
{

    {
        var summary = BenchmarkRunner.Run<EfCoreBenchmarks>();
