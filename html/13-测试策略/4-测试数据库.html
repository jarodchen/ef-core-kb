1. 测试数据库选型 ​

1.1 选型决策矩阵 ​

方案速度真实性复杂度跨平台CI/CD友好推荐场景
SQLite内存⚡⚡⚡⚡⚡⭐⭐⭐🟢低✅✅单元测试
InMemory⚡⚡⚡⚡⚡⭐⭐🟢低✅✅快速原型

2. SQLite内存模式 ​

2.1 为什么选择SQLite? ​

优势:

✅ 超快速度 - 内存操作,毫秒级启动
✅ 零配置 - 无需安装,开箱即用
✅ 完全隔离 - 每个测试独立数据库
✅ 跨平台 - Windows/Linux/macOS通用
✅ 真实SQL - 比InMemory更接近真实行为

局限性:

⚠️ SQL方言差异 - 不完全兼容T-SQL
⚠️ 并发限制 - 单线程访问
⚠️ 功能缺失 - 不支持某些高级特性

2.2 完整实现 ​

测试基类:

csharp
public abstract class SqliteTestBase : IDisposable
{
    protected readonly AppDbContext Context;
    private readonly DbConnection _connection;
    
    protected SqliteTestBase()
    {
        // 创建并打开连接
        _connection = new SqliteConnection("Filename=:memory:");
        _connection.Open();
        
        // 配置DbContext
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlite(_connection)
            .EnableSensitiveDataLogging()  // 调试用
            .LogTo(Console.WriteLine, LogLevel.Information)  // 输出SQL
            .Options;
        
        Context = new AppDbContext(options);
        
        // 创建数据库架构
        Context.Database.EnsureCreated();
        
        // 可选:应用迁移而非EnsureCreated
        // Context.Database.Migrate();
    }
    
    /// <summary>
    /// 种子数据辅助方法
    /// </summary>
    protected async Task SeedAsync(params object[] entities)
    {
        Context.AddRange(entities);
        await Context.SaveChangesAsync();
    }
    
    public void Dispose()
    {
        Context.Dispose();
        _connection.Close();
        _connection.Dispose();
    }
}

2.3 处理SQLite兼容性 ​

常见兼容性问题:

csharp
public class SqliteCompatibilityTests
{
    [Fact]
    public async Task HandleDateTimePrecision()
    {
        // SQLite的DateTime精度与SQL Server不同
        var order = new Order 
        { 
            OrderDate = DateTime.UtcNow.AddTicks(12345)  // 高精度
        };
        
        Context.Orders.Add(order);
        await Context.SaveChangesAsync();
        
        var loaded = await Context.Orders.FirstAsync();
        
        // ⚠️ 可能不相等(精度丢失)
        Assert.NotEqual(order.OrderDate, loaded.OrderDate);
        
        // ✅ 使用近似比较
        Assert.True(Math.Abs((order.OrderDate - loaded.OrderDate).TotalMilliseconds) < 1);
    }
    
    [Fact]
    public async Task HandleCaseSensitivity()
    {
        // SQLite默认大小写敏感(SQL Server不敏感)
        var product = new Product { Name = "LAPTOP" };
        Context.Products.Add(product);
        await Context.SaveChangesAsync();
        
        // SQL Server: 能找到(LAPTOP = laptop)
        // SQLite: 找不到(大小写敏感)
        var found = await Context.Products
            .FirstOrDefaultAsync(p => p.Name == "laptop");
        
        Assert.Null(found);  // SQLite返回null
        
        // ✅ 解决方案: 使用COLLATE
        var found2 = await Context.Products
            .FirstOrDefaultAsync(p => 
                EF.Functions.Like(p.Name, "laptop"));  // 不区分大小写
    }
}

3.2 配置LocalDB ​

基础配置:

csharp
public class LocalDbTestBase : IDisposable
{
    protected readonly AppDbContext Context;
    private readonly string _databaseName;
    
    protected LocalDbTestBase()
    {
        // 唯一数据库名(避免冲突)
        _databaseName = $"TestDb_{Guid.NewGuid():N}";
        
        var connectionString = $@"
            Server=(localdb)\mssqllocaldb;
            Database={_databaseName};
            Trusted_Connection=True;
            MultipleActiveResultSets=true;";
        
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlServer(connectionString, sqlOptions =>
            {
                sqlOptions.EnableRetryOnFailure(
                    maxRetryCount: 3,
                    maxRetryDelay: TimeSpan.FromSeconds(5),
                    errorNumbersToAdd: null);
            })
            .Options;
        
        Context = new AppDbContext(options);
        
        // 创建数据库
        Context.Database.EnsureCreated();
    }
    
    public void Dispose()
    {
        Context.Dispose();
        
        // 删除数据库(清理)
        using var cleanupContext = new AppDbContext(
            new DbContextOptionsBuilder<AppDbContext>()
                .UseSqlServer($@"Server=(localdb)\mssqllocaldb;Database={_databaseName};Trusted_Connection=True;")
                .Options);
        
        cleanupContext.Database.EnsureDeleted();
    }
}

4. Docker容器化方案 ​

4.1 Testcontainers(推荐) ​

完整实现:

csharp
public class TestcontainersFixture : IAsyncLifetime
{
    private readonly MsSqlContainer _container;
    
    public string ConnectionString { get; private set; }
    
    public TestcontainersFixture()
    {
        _container = new MsSqlBuilder()
            .WithImage("mcr.microsoft.com/mssql/server:2022-latest")
            .WithName($"efcore-tests-{Guid.NewGuid():N}")
            .WithPassword("YourStrong@Passw0rd")
            .WithPortBinding(1433, assignRandomHostPort: true)
            .WithCleanUp(true)
            .Build();
    }
    
    public async Task InitializeAsync()
    {
        await _container.StartAsync();
        ConnectionString = _container.GetConnectionString();
        
        // 等待数据库就绪
        await WaitForDatabaseReadyAsync(ConnectionString);
        
        // 应用迁移
        await ApplyMigrationsAsync(ConnectionString);
    }
    
    private async Task WaitForDatabaseReadyAsync(string connectionString, int timeoutSeconds = 30)
    {
        var stopwatch = Stopwatch.StartNew();
        
        while (stopwatch.Elapsed.TotalSeconds < timeoutSeconds)
        {
            try
            {
                using var connection = new SqlConnection(connectionString);
                await connection.OpenAsync();
                
                using var command = new SqlCommand("SELECT 1", connection);
                await command.ExecuteScalarAsync();
                
                return;  // 就绪
            }
            catch
            {
                await Task.Delay(500);  // 等待重试
            }
        }
        
        throw new TimeoutException("Database not ready within timeout");
    }
    
    private async Task ApplyMigrationsAsync(string connectionString)
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlServer(connectionString)
            .Options;
        
        using var context = new AppDbContext(options);
        await context.Database.MigrateAsync();
    }
    
    public async Task DisposeAsync()
    {
        await _container.DisposeAsync();
    }
}

策略2: 重用数据库(更快)

csharp
public class ReuseDatabaseFixture : IAsyncLifetime
{
    private static bool _initialized;
    private static readonly SemaphoreSlim _semaphore = new(1, 1);
    
    public string ConnectionString { get; private set; }
    
    public async Task InitializeAsync()
    {
        ConnectionString = "Server=(localdb)\\mssqllocaldb;Database=SharedTestDb;";
        
        await _semaphore.WaitAsync();
        try
        {
            if (!_initialized)
            {
                var options = new DbContextOptionsBuilder<AppDbContext>()
                    .UseSqlServer(ConnectionString)
                    .Options;
                
                await using var context = new AppDbContext(options);
                await context.Database.EnsureDeletedAsync();
                await context.Database.MigrateAsync();
                
                _initialized = true;
            }
        }
        finally
        {
            _semaphore.Release();
        }
    }
    
    public Task DisposeAsync()
    {
        // 不删除,留给下次测试
        return Task.CompletedTask;
    }
}

6.2 数据清理辅助类 ​

csharp
public static class DatabaseCleaner
{
    public static async Task CleanAllTablesAsync(AppDbContext context)
    {
        // 禁用外键约束
        await context.Database.ExecuteSqlRawAsync("EXEC sp_MSforeachtable 'ALTER TABLE ? NOCHECK CONSTRAINT ALL'");
        
        // 删除所有数据
        var tables = context.Model.GetEntityTypes()
            .Select(t => t.GetTableName())
            .Distinct();
        
        foreach (var table in tables)
        {
            await context.Database.ExecuteSqlRawAsync($"DELETE FROM [{table}]");
        }
        
        // 重置自增ID
        await context.Database.ExecuteSqlRawAsync("EXEC sp_MSforeachtable 'DBCC CHECKIDENT (''?'' , RESEED, 0)'");
        
        // 启用外键约束
        await context.Database.ExecuteSqlRawAsync("EXEC sp_MSforeachtable 'ALTER TABLE ? WITH CHECK CHECK CONSTRAINT ALL'");
    }
}

public class TransactionIsolationTests : IDisposable
{
    private readonly AppDbContext _context;
    private readonly IDbContextTransaction _transaction;
    
    public TransactionIsolationTests()
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlServer(ConnectionString)
            .Options;
        
        _context = new AppDbContext(options);
        _transaction = _context.Database.BeginTransaction();  // 每个测试独立事务
    }
    
    [Fact]
    public async Task Test1_Isolated()
    {
        // 数据在此事务内
        _context.Products.Add(new Product { Name = "Test1" });
        await _context.SaveChangesAsync();
        
        // 其他测试看不到此数据
    }
    
    public void Dispose()
    {
        _transaction.Rollback();  // 回滚,清除所有更改
        _transaction.Dispose();
        _context.Dispose();
    }
}

8.2 监控查询性能 ​

csharp
public class QueryPerformanceMonitor
{
    private readonly ILogger<QueryPerformanceMonitor> _logger;
    
    public async Task MonitorQueryAsync(Func<AppDbContext, Task> queryAction)
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlServer(ConnectionString)
            .LogTo(
                message => 
                {
                    if (message.Contains("Executed DbCommand"))
                    {
                        _logger.LogInformation("SQL: {Message}", message);
                    }
                },
                LogLevel.Information
            )
            .Options;
        
        await using var context = new AppDbContext(options);
        
        var stopwatch = Stopwatch.StartNew();
        await queryAction(context);
        stopwatch.Stop();
        
        _logger.LogInformation("Query completed in {ElapsedMs}ms", stopwatch.ElapsedMilliseconds);
    }
}

总结 ​

测试数据库策略是EF Core质量保证的核心:

推荐方案组合 ​

单元测试 → SQLite内存模式
集成测试 → Testcontainers(Docker)
CI/CD → Docker Compose
E2E测试 → 云数据库

关键原则 ​

✅ 隔离性 - 每个测试独立
✅ 可重复 - Deterministic结果
✅ 快速反馈 - 秒级执行
✅ 真实性 - 接近生产环境
✅ 易维护 - 自动化清理

掌握这些技术,你可以构建高效可靠的测试基础设施!
