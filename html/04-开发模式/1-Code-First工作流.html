Code First(代码优先)工作流 ​

Code First是EF Core最流行的开发模式,允许开发者通过C#类定义数据模型,然后由EF Core自动创建和维护数据库架构。本章将深入探讨Code First的完整工作流程,包括实体设计、迁移管理、数据库初始化以及最佳实践。

1. Code First基础 ​

public class AppDbContext : DbContext

public class AppDbContext : DbContext
{
    public AppDbContext(DbContextOptions<AppDbContext> options)
        : base(options)
    {

    public DbSet<Product> Products { get; set; }

        // 配置实体

            entity.HasIndex(p => p.Name);
        });
    }
}

2.1 约定优于配置 ​

EF Core默认约定:

csharp
public class Product
{
    // ✅ 约定: 名为Id或{ClassName}Id的属性自动成为主键
    public int Id { get; set; }
    
    // ✅ 约定: 字符串映射为nvarchar(max)
    public string Name { get; set; }
    
    // ✅ 约定: decimal映射为decimal(18,2)
    public decimal Price { get; set; }
    
    // ✅ 约定: 可空类型允许NULL
    public DateTime? DiscontinuedAt { get; set; }
    
    // ✅ 约定: 集合属性自动成为导航属性
    public List<OrderItem> OrderItems { get; set; }

2.2 Fluent API配置 ​

完整配置示例:

csharp
protected override void OnModelCreating(ModelBuilder modelBuilder)
{
    // Product配置
    modelBuilder.Entity<Product>(entity =>
    {
        // 主键
        entity.HasKey(p => p.Id);
        
        // 表名
        entity.ToTable("Products", "dbo");
        
        // 属性配置
        entity.Property(p => p.Name)
            .IsRequired()
            .HasMaxLength(200)
            .HasColumnName("ProductName");
        
        entity.Property(p => p.Price)
            .HasColumnType("decimal(18,2)")
            .HasDefaultValue(0m);
        
        entity.Property(p => p.Stock)
            .HasDefaultValue(0)
            .ValueGeneratedOnAdd();
        
        // 索引
        entity.HasIndex(p => p.Name)
            .IsUnique();
        
        entity.HasIndex(p => p.Price);
        
        // 关系配置
        entity.HasMany(p => p.OrderItems)
            .WithOne(oi => oi.Product)
            .HasForeignKey(oi => oi.ProductId)
            .OnDelete(DeleteBehavior.Restrict);
    });
}

csharp
[Migration("20240115103000_AddProductCategory")]
public partial class AddProductCategory : Migration
{
    protected override void Up(MigrationBuilder migrationBuilder)
    {
        // 添加分类表
        migrationBuilder.CreateTable(
            name: "Categories",
            columns: table => new
            {
                Id = table.Column<int>(nullable: false)
                    .Annotation("SqlServer:Identity", "1, 1"),
                Name = table.Column<string>(maxLength: 100, nullable: false)
            },
            constraints: table =>
            {
                table.PrimaryKey("PK_Categories", x => x.Id);
            });
        
        // 为产品表添加外键
        migrationBuilder.AddColumn<int>(
            name: "CategoryId",
            table: "Products",
            nullable: false,
            defaultValue: 1);
        
        migrationBuilder.CreateIndex(
            name: "IX_Products_CategoryId",
            table: "Products",
            column: "CategoryId");
        
        migrationBuilder.AddForeignKey(
            name: "FK_Products_Categories_CategoryId",
            table: "Products",
            column: "CategoryId",
            principalTable: "Categories",
            principalColumn: "Id",
            onDelete: ReferentialAction.Restrict);
    }

    protected override void Down(MigrationBuilder migrationBuilder)
    {
        migrationBuilder.DropForeignKey(
            name: "FK_Products_Categories_CategoryId",
            table: "Products");
        
        migrationBuilder.DropIndex(
            name: "IX_Products_CategoryId",
            table: "Products");
        
        migrationBuilder.DropColumn(
            name: "CategoryId",
            table: "Products");
        
        migrationBuilder.DropTable(
            name: "Categories");
    }
}

4. 数据库初始化策略 ​

4.1 EnsureCreated(仅开发) ​

csharp
// Program.cs - 仅用于开发/测试
using var scope = app.Services.CreateScope();
var context = scope.ServiceProvider.GetRequiredService<AppDbContext>();

// 如果数据库不存在则创建(不使用迁移)
if (context.Database.EnsureCreated())
{
    Console.WriteLine("Database created successfully");
}
else
{
    Console.WriteLine("Database already exists");
}

// ⚠️ 警告: EnsureCreated不与迁移配合使用!

4.3 条件初始化 ​

csharp
public static class DatabaseInitializer
{
    public static async Task InitializeAsync(AppDbContext context)
    {
        // 检查是否有待处理的迁移
        var pendingMigrations = await context.Database.GetPendingMigrationsAsync();
        
        if (pendingMigrations.Any())
        {
            Console.WriteLine($"Applying {pendingMigrations.Count()} migrations...");
            await context.Database.MigrateAsync();
        }
        else
        {
            Console.WriteLine("Database is up to date");
        }
        
        // 可选: 种子数据
        await SeedDataAsync(context);
    }
    
    private static async Task SeedDataAsync(AppDbContext context)
    {
        if (!context.Categories.Any())
        {
            context.Categories.AddRange(
                new Category { Name = "Electronics" },
                new Category { Name = "Books" }
            );
            
            await context.SaveChangesAsync();
        }
    }
}

// 使用
await DatabaseInitializer.InitializeAsync(context);

5.2 运行时种子数据 ​

csharp
public static class DataSeeder
{
    public static async Task SeedAsync(AppDbContext context)
    {
        // 检查是否已有数据
        if (await context.Categories.AnyAsync())
            return;
        
        // 种子数据
        var categories = new List<Category>
        {
            new() { Name = "Electronics" },
            new() { Name = "Books" }
        };
        
        await context.Categories.AddRangeAsync(categories);
        await context.SaveChangesAsync();
        
        var products = new List<Product>
        {
            new() 
            { 
                Name = "Laptop", 
                Price = 999.99m,
                CategoryId = categories[0].Id 
            },
            new() 
            { 
                Name = "Phone", 
                Price = 699.99m,
                CategoryId = categories[0].Id 
            }
        };
        
        await context.Products.AddRangeAsync(products);
        await context.SaveChangesAsync();
    }
}

// 在Program.cs中调用
await DataSeeder.SeedAsync(context);

6.1 重命名属性 ​

csharp
// 原实体
public class Product
{
    public int Id { get; set; }
    public string ProductName { get; set; }  // 想改为Name

// 修改实体
public class Product
{
    public int Id { get; set; }
    public string Name { get; set; }  // 重命名

// 手动编辑迁移文件
public partial class RenameProductNameToName : Migration
{
    protected override void Up(MigrationBuilder migrationBuilder)
    {
        migrationBuilder.RenameColumn(
            name: "ProductName",
            table: "Products",
            newName: "Name");
    }

    protected override void Down(MigrationBuilder migrationBuilder)
    {
        migrationBuilder.RenameColumn(
            name: "Name",
            table: "Products",
            newName: "ProductName");
    }
}

6.2 拆分实体 ​

csharp
// 原实体
public class Customer
{
    public int Id { get; set; }
    public string Name { get; set; }
    public string Email { get; set; }
    public string Street { get; set; }      // 要拆分
    public string City { get; set; }        // 要拆分
    public string ZipCode { get; set; }     // 要拆分
}

// 重构后
public class Customer
{
    public int Id { get; set; }
    public string Name { get; set; }
    public string Email { get; set; }
    public Address Address { get; set; }  // 值对象
}

public class Address
{
    public string Street { get; set; }
    public string City { get; set; }
    public string ZipCode { get; set; }
}

// 配置
modelBuilder.Entity<Customer>(entity =>
{
    entity.OwnsOne(c => c.Address, address =>
    {
        address.Property(a => a.Street).HasColumnName("Street");
        address.Property(a => a.City).HasColumnName("City");
        address.Property(a => a.ZipCode).HasColumnName("ZipCode");
    });
});

总结 ​

Code First是EF Core推荐的开发模式:

核心优势 ​

✅ 代码驱动 - 完全控制模型
✅ 易于重构 - 迁移支持演进
✅ 团队协作 - Git友好

关键工作流 ​

  1. 设计实体 - 定义领域模型
  2. 配置映射 - Fluent API
  3. 创建迁移 - dotnet ef migrations add
  4. 应用迁移 - dotnet ef database update
  5. 迭代演进 - 持续重构

掌握Code First,你可以高效地开发和维护数据访问层!
