多租户架构(Multi-Tenancy) ​

多租户架构是一种软件架构模式,允许单个应用实例服务于多个客户(租户),每个租户的数据完全隔离。本章将深入探讨在 EF Core 中实现多租户的多种策略,包括数据库级别隔离、Schema隔离、行级别隔离,以及租户发现、数据过滤和性能优化。

1. 多租户基础 ​

1.1 什么是多租户? ​

核心概念:

csharp
// 单租户架构 vs 多租户架构

// ❌ 单租户: 每个客户独立部署
Customer A → App Instance A → Database A
Customer B → App Instance B → Database B
Customer C → App Instance C → Database C
// 问题: 资源浪费,维护成本高

// ✅ 多租户: 共享应用实例,数据隔离
Customer A ──┐
Customer B ──┼──→ Single App Instance → Isolated Data
Customer C ──┘
// 优势: 资源共享,成本低,易维护

多租户的优势:

✅ 成本效益 - 共享基础设施,降低运营成本
✅ 资源利用 - 动态分配资源,提高利用率
✅ 易于维护 - 单一代码库,统一升级
✅ 可扩展性 - 轻松添加新租户
✅ 数据隔离 - 租户间数据安全隔离

典型应用场景:

  • SaaS应用(Software as a Service)
  • 云平台服务
  • ERP/CRM系统
  • 电商平台(多商家)
  • 协作工具(多团队)

1.2 三种隔离级别对比 ​

策略隔离度成本复杂度适用场景
数据库隔离⭐⭐⭐⭐⭐高中大客户、强合规
Schema隔离⭐⭐⭐⭐中高中型客户

3. 行级别隔离实现 ​

3.1 全局查询过滤器(推荐) ​

自动应用租户过滤:

csharp
public class MultiTenantDbContext : DbContext
{
    private readonly Guid _currentTenantId;
    
    public MultiTenantDbContext(DbContextOptions options, Guid currentTenantId)
        : base(options)
    {
        _currentTenantId = currentTenantId;
    }
    
    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);
        
        // 为所有实现 IMultiTenant 的实体添加全局过滤器
        var multiTenantTypes = modelBuilder.Model.GetEntityTypes()
            .Where(t => typeof(IMultiTenant).IsAssignableFrom(t.ClrType));
        
        foreach (var entityType in multiTenantTypes)
        {
            // 动态创建 lambda: e => e.TenantId == _currentTenantId
            var parameter = Expression.Parameter(entityType.ClrType, "e");
            var property = Expression.Property(parameter, nameof(IMultiTenant.TenantId));
            var tenantIdValue = Expression.Constant(_currentTenantId);
            var equalExpression = Expression.Equal(property, tenantIdValue);
            var lambda = Expression.Lambda(equalExpression, parameter);
            
            modelBuilder.Entity(entityType.ClrType).HasQueryFilter(lambda);
        }
    }
    
    public override async Task<int> SaveChangesAsync(CancellationToken cancellationToken = default)
    {
        // 自动设置 TenantId
        foreach (var entry in ChangeTracker.Entries<IMultiTenant>())
        {
            if (entry.State == EntityState.Added)
            {
                entry.Entity.TenantId = _currentTenantId;
            }
        }
        
        return await base.SaveChangesAsync(cancellationToken);
    }
}

3.2 租户拦截器 ​

自动设置租户信息:

csharp
public class TenantInterceptor : SaveChangesInterceptor
{
    private readonly ITenantProvider _tenantProvider;
    
    public TenantInterceptor(ITenantProvider tenantProvider)
    {
        _tenantProvider = tenantProvider;
    }
    
    public override async ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        var context = eventData.Context;
        if (context == null) return result;
        
        var currentTenantId = _tenantProvider.GetCurrentTenantId();
        
        foreach (var entry in context.ChangeTracker.Entries())
        {
            if (entry.Entity is IMultiTenant multiTenant)
            {
                switch (entry.State)
                {
                    case EntityState.Added:
                        // 自动设置租户ID
                        multiTenant.TenantId = currentTenantId;
                        
                        // 设置审计字段
                        if (entry.Entity is MultiTenantEntity auditable)
                        {
                            auditable.CreatedAt = DateTime.UtcNow;
                            auditable.CreatedBy = _tenantProvider.GetCurrentUserId();
                        }
                        break;
                    
                    case EntityState.Modified:
                        if (entry.Entity is MultiTenantEntity auditable2)
                        {
                            auditable2.UpdatedAt = DateTime.UtcNow;
                            auditable2.UpdatedBy = _tenantProvider.GetCurrentUserId();
                        }
                        break;
                }
            }
        }
        
        return await base.SavingChangesAsync(eventData, result, cancellationToken);
    }
}

// 租户提供者
public interface ITenantProvider
{
    Guid GetCurrentTenantId();
    string? GetCurrentUserId();
}

public class HttpContextTenantProvider : ITenantProvider
{
    private readonly IHttpContextAccessor _httpContextAccessor;
    private readonly ITenantResolver _tenantResolver;
    
    public HttpContextTenantProvider(
        IHttpContextAccessor httpContextAccessor,
        ITenantResolver tenantResolver)
    {
        _httpContextAccessor = httpContextAccessor;
        _tenantResolver = tenantResolver;
    }
    
    public Guid GetCurrentTenantId()
    {
        var subdomain = GetSubdomain();
        return _tenantResolver.ResolveTenantId(subdomain);
    }
    
    public string? GetCurrentUserId()
    {
        return _httpContextAccessor.HttpContext?.User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
    }
    
    private string GetSubdomain()
    {
        var host = _httpContextAccessor.HttpContext?.Request.Host.Host;
        if (string.IsNullOrEmpty(host))
            throw new InvalidOperationException("Cannot determine subdomain");
        
        // tenant1.example.com -> tenant1
        return host.Split('.')[0];
    }
}

为租户查询优化性能:

csharp
protected override void OnModelCreating(ModelBuilder modelBuilder)
{
    // Product 表
    modelBuilder.Entity<Product>(entity =>
    {
        // 租户ID + 主键的复合索引
        entity.HasIndex(p => new { p.TenantId, p.Id });
        
        // 常用查询的复合索引
        entity.HasIndex(p => new { p.TenantId, p.Name });
        entity.HasIndex(p => new { p.TenantId, p.Price });
        entity.HasIndex(p => new { p.TenantId, p.CreatedAt });
        
        // 唯一约束(租户内唯一)
        entity.HasIndex(p => new { p.TenantId, p.Name }).IsUnique();
    });
    
    // Order 表
    modelBuilder.Entity<Order>(entity =>
    {
        entity.HasIndex(o => new { o.TenantId, o.OrderDate });
        entity.HasIndex(o => new { o.TenantId, o.CustomerId });
    });
}

4.1 动态Schema切换 ​

为每个租户使用独立Schema:

csharp
public class SchemaMultiTenantDbContext : DbContext
{
    private readonly Tenant _currentTenant;
    
    public SchemaMultiTenantDbContext(DbContextOptions options, Tenant currentTenant)
        : base(options)
    {
        _currentTenant = currentTenant;
    }
    
    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);
        
        // 为所有实体设置Schema
        foreach (var entityType in modelBuilder.Model.GetEntityTypes())
        {
            if (entityType.ClrType.Namespace?.StartsWith("MyApp.Models") == true)
            {
                entityType.SetSchema(_currentTenant.SchemaName);
            }
        }
    }
}

// 租户解析中间件
public class TenantResolutionMiddleware
{
    private readonly RequestDelegate _next;
    
    public TenantResolutionMiddleware(RequestDelegate next)
    {
        _next = next;
    }
    
    public async Task InvokeAsync(HttpContext context, ITenantResolver resolver)
    {
        var subdomain = GetSubdomain(context);
        var tenant = await resolver.GetTenantBySubdomainAsync(subdomain);
        
        if (tenant == null || !tenant.IsActive)
        {
            context.Response.StatusCode = 404;
            await context.Response.WriteAsync("Tenant not found");
            return;
        }
        
        // 存储到HttpContext
        context.Items["CurrentTenant"] = tenant;
        
        await _next(context);
    }
    
    private string GetSubdomain(HttpContext context)
    {
        var host = context.Request.Host.Host;
        return host.Split('.')[0];
    }
}

// Program.cs
app.UseMiddleware<TenantResolutionMiddleware>();

csharp
public class TenantMigrationService
{
    private readonly IServiceProvider _serviceProvider;
    
    public async Task MigrateAllTenantsAsync()
    {
        var tenants = await GetAllActiveTenantsAsync();
        
        foreach (var tenant in tenants)
        {
            try
            {
                await MigrateTenantAsync(tenant);
                Console.WriteLine($"Migrated tenant: {tenant.Name}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Failed to migrate tenant {tenant.Name}: {ex.Message}");
            }
        }
    }
    
    private async Task MigrateTenantAsync(Tenant tenant)
    {
        var options = new DbContextOptionsBuilder<SchemaMultiTenantDbContext>()
            .UseSqlServer(tenant.ConnectionString)
            .Options;
        
        await using var context = new SchemaMultiTenantDbContext(options, tenant);
        await context.Database.MigrateAsync();
    }
}

public class DatabasePerTenantDbContext : DbContext

// 租户数据库工厂
public class TenantDbContextFactory
{
    private readonly ITenantProvider _tenantProvider;
    
    public DatabasePerTenantDbContext Create()
    {
        var tenantId = _tenantProvider.GetCurrentTenantId();
        var connectionString = GetConnectionStringForTenant(tenantId);
        
        var options = new DbContextOptionsBuilder<DatabasePerTenantDbContext>()
            .UseSqlServer(connectionString)
            .Options;
        
        return new DatabasePerTenantDbContext(options);
    }
    
    private string GetConnectionStringForTenant(Guid tenantId)
    {
        // 从配置或数据库获取租户的连接字符串
        var tenant = await GetTenantAsync(tenantId);
        return tenant.ConnectionString 
            ?? throw new InvalidOperationException($"No connection string for tenant {tenantId}");
    }
}

5.2 租户数据库 provisioning ​

自动创建租户数据库:

csharp
public class TenantProvisioningService
{
    private readonly string _masterConnectionString;
    
    public async Task<Tenant> CreateTenantAsync(CreateTenantRequest request)
    {
        var tenant = new Tenant
        {
            Id = Guid.NewGuid(),
            Name = request.Name,
            Subdomain = request.Subdomain,
            Plan = request.Plan,
            CreatedAt = DateTime.UtcNow,
            IsActive = true
        };
        
        // 1. 创建数据库
        var databaseName = $"TenantDB_{tenant.Id:N}";
        await CreateDatabaseAsync(databaseName);
        
        // 2. 生成连接字符串
        tenant.ConnectionString = BuildConnectionString(databaseName);
        
        // 3. 应用迁移
        await ApplyMigrationsAsync(tenant.ConnectionString);
        
        // 4. 种子数据
        await SeedDefaultDataAsync(tenant.ConnectionString);
        
        // 5. 保存租户信息
        await SaveTenantAsync(tenant);
        
        return tenant;
    }
    
    private async Task CreateDatabaseAsync(string databaseName)
    {
        await using var connection = new SqlConnection(_masterConnectionString);
        await connection.OpenAsync();
        
        var command = connection.CreateCommand();
        command.CommandText = $"CREATE DATABASE [{databaseName}]";
        await command.ExecuteNonQueryAsync();
    }
    
    private async Task ApplyMigrationsAsync(string connectionString)
    {
        var options = new DbContextOptionsBuilder<DatabasePerTenantDbContext>()
            .UseSqlServer(connectionString)
            .Options;
        
        await using var context = new DatabasePerTenantDbContext(options);
        await context.Database.MigrateAsync();
    }
}

// X-Tenant-ID: guid
public class HeaderTenantResolver : ITenantResolver
{
    public Guid ResolveTenantId(HttpContext context)
    {
        var tenantIdHeader = context.Request.Headers["X-Tenant-ID"].FirstOrDefault();
        
        if (string.IsNullOrEmpty(tenantIdHeader) || !Guid.TryParse(tenantIdHeader, out var tenantId))
            throw new InvalidOperationException("Invalid or missing X-Tenant-ID header");
        
        return tenantId;
    }
}

// /api/tenant1/products
public class PathTenantResolver : ITenantResolver
{
    public Guid ResolveTenantId(HttpContext context)
    {
        var segments = context.Request.Path.Value?.Split('/', StringSplitOptions.RemoveEmptyEntries);
        
        if (segments == null || segments.Length < 2 || segments[0] != "api")
            throw new InvalidOperationException("Invalid path format");
        
        var subdomain = segments[1];

csharp
public class CachedTenantResolver : ITenantResolver
{
    private readonly IMemoryCache _cache;
    private readonly ITenantRepository _repository;
    private readonly TimeSpan _cacheDuration = TimeSpan.FromMinutes(30);
    
    public async Task<Tenant> GetTenantAsync(Guid tenantId)
    {
        var cacheKey = $"tenant:{tenantId}";
        
        return await _cache.GetOrCreateAsync(cacheKey, async entry =>
        {
            entry.AbsoluteExpirationRelativeToNow = _cacheDuration;
            
            var tenant = await _repository.GetByIdAsync(tenantId);
            
            if (tenant == null || !tenant.IsActive)
                throw new NotFoundException($"Tenant {tenantId} not found or inactive");
            
            return tenant;
        });
    }
    
    public async Task InvalidateCacheAsync(Guid tenantId)
    {
        _cache.Remove($"tenant:{tenantId}");
    }
}

public class AdminDashboardService
{
    private readonly MasterDbContext _masterContext;  // 主数据库上下文
    
    /// <summary>
    /// 平台级统计(跨所有租户)
    /// </summary>
    public async Task<PlatformStats> GetPlatformStatsAsync()
    {
        var tenants = await _masterContext.Tenants
            .Where(t => t.IsActive)
            .ToListAsync();
        
        var stats = new PlatformStats();
        
        foreach (var tenant in tenants)
        {
            var tenantStats = await GetTenantStatsAsync(tenant);
            stats.TotalOrders += tenantStats.OrderCount;
            stats.TotalRevenue += tenantStats.Revenue;
            stats.TotalUsers += tenantStats.UserCount;
        }
        
        return stats;
    }
    
    private async Task<TenantStats> GetTenantStatsAsync(Tenant tenant)
    {
        // 切换到租户数据库
        var options = new DbContextOptionsBuilder<TenantDbContext>()
            .UseSqlServer(tenant.ConnectionString)
            .Options;
        
        await using var context = new TenantDbContext(options);
        
        return new TenantStats
        {
            TenantId = tenant.Id,
            OrderCount = await context.Orders.CountAsync(),
            Revenue = await context.Orders.SumAsync(o => o.TotalAmount),
            UserCount = await context.Users.CountAsync()
        };
    }
}
