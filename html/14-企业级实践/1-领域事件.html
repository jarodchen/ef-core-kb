领域事件(Domain Events) ​

领域事件是领域驱动设计(DDD)中的核心模式,用于捕获领域中发生的重要业务事件。本章将深入探讨如何在 EF Core 应用中实现领域事件模式,包括同步和异步事件处理、事件存储、以及与企业服务总线的集成。

1. 领域事件基础 ​

1.1 什么是领域事件? ​

定义: 领域事件表示在业务领域中发生的重要事情,通常是过去时态的事实。

特征:

  • ✅ 过去时态 - OrderCreated, PaymentProcessed, InventoryDeducted
  • ✅ 不可变性 - 发生后不能修改
  • ✅ 携带数据 - 包含事件发生时的上下文信息
  • ✅ 解耦性 - 发布者和订阅者松耦合

典型应用场景:

csharp
// 电商系统示例
OrderCreatedEvent          // 订单创建
PaymentProcessedEvent      // 支付处理完成
InventoryDeductedEvent     // 库存扣减
OrderShippedEvent          // 订单发货
OrderDeliveredEvent        // 订单送达
CustomerRegisteredEvent    // 客户注册
ProductPriceChangedEvent   // 产品价格变更

// 金融系统示例
AccountOpenedEvent         // 账户开立
MoneyTransferredEvent      // 资金转账
LoanApprovedEvent          // 贷款审批

// 协作系统示例
TaskAssignedEvent          // 任务分配
CommentAddedEvent          // 评论添加
ProjectCreatedEvent        // 项目创建

1.2 领域事件 vs 域内通知 ​

csharp
// ❌ 方式1: 紧耦合的域内通知
public class OrderService
{
    private readonly EmailService _emailService;
    private readonly InventoryService _inventoryService;
    private readonly AnalyticsService _analyticsService;
    
    public async Task CreateOrder(Order order)
    {
        // 创建订单
        await _orderRepository.AddAsync(order);
        
        // 紧耦合: 直接调用所有相关服务
        await _emailService.SendConfirmation(order.CustomerId, order.Id);
        await _inventoryService.DeductStock(order.Items);
        await _analyticsService.TrackOrder(order);
        await _notificationService.NotifyWarehouse(order);
    }
}

// ✅ 方式2: 使用领域事件解耦
public class OrderService
{
    private readonly IEventBus _eventBus;
    
    public async Task CreateOrder(Order order)
    {
        // 创建订单
        await _orderRepository.AddAsync(order);
        
        // 发布领域事件 - 不关心谁订阅
        await _eventBus.PublishAsync(new OrderCreatedEvent(
            order.Id,
            order.CustomerId,
            order.Items,
            order.TotalAmount
        ));
    }
}

// 各个订阅者独立处理
public class OrderConfirmationHandler : IEventHandler<OrderCreatedEvent>
{
    public async Task Handle(OrderCreatedEvent @event)
    {
        await _emailService.SendConfirmation(@event.CustomerId, @event.OrderId);
    }
}

public class InventoryDeductionHandler : IEventHandler<OrderCreatedEvent>
{
    public async Task Handle(OrderCreatedEvent @event)
    {
        await _inventoryService.DeductStock(@event.Items);
    }
}

public abstract class DomainEvent : IDomainEvent
{

实体中收集事件:

csharp
public abstract class AggregateRoot
{
    private readonly List<IDomainEvent> _domainEvents = new();
    
    public IReadOnlyCollection<IDomainEvent> DomainEvents => 
        _domainEvents.AsReadOnly();
    
    protected void AddDomainEvent(IDomainEvent domainEvent)
    {
        _domainEvents.Add(domainEvent);
    }
    
    public void ClearDomainEvents()
    {
        _domainEvents.Clear();
    }
    
    protected void RemoveDomainEvent(IDomainEvent domainEvent)
    {
        _domainEvents.Remove(domainEvent);
    }
}

// 聚合根示例
public class Order : AggregateRoot
{
    public int Id { get; private set; }
    public string CustomerId { get; private set; }
    public DateTime OrderDate { get; private set; }
    public OrderStatus Status { get; private set; }
    public decimal TotalAmount { get; private set; }
    
    private readonly List<OrderItem> _items = new();
    public IReadOnlyCollection<OrderItem> Items => _items.AsReadOnly();
    
    // 私有构造函数 - 强制使用工厂方法
    private Order() { }
    
    // 工厂方法 - 创建订单并发布事件
    public static Order Create(string customerId, List<OrderItem> items, 
        string shippingAddress)
    {
        var order = new Order
        {
            CustomerId = customerId,
            OrderDate = DateTime.UtcNow,
            Status = OrderStatus.Pending,
            _items = items,
            TotalAmount = items.Sum(i => i.Quantity * i.UnitPrice)
        };
        
        // 添加领域事件
        order.AddDomainEvent(new OrderCreatedEvent(
            order.Id,
            order.CustomerId,
            items.Select(i => new OrderItemDto(i.ProductId, i.Quantity, i.UnitPrice)).ToList(),
            order.TotalAmount,
            shippingAddress
        ));
        
        return order;
    }
    
    // 业务方法 - 确认订单
    public void Confirm()
    {
        if (Status != OrderStatus.Pending)
            throw new InvalidOperationException("Order is not pending");
        
        Status = OrderStatus.Confirmed;
        
        // 添加领域事件
        AddDomainEvent(new OrderConfirmedEvent(Id, CustomerId, TotalAmount));
    }
    
    // 业务方法 - 取消订单
    public void Cancel(string reason)
    {
        if (Status == OrderStatus.Cancelled)
            throw new InvalidOperationException("Order already cancelled");
        
        var previousStatus = Status;
        Status = OrderStatus.Cancelled;
        
        AddDomainEvent(new OrderCancelledEvent(Id, CustomerId, previousStatus, reason));
    }
}

2.2 事件总线接口 ​

csharp
// 事件总线抽象
public interface IEventBus
{
    /// <summary>
    /// 发布事件(异步)
    /// </summary>
    Task PublishAsync<TEvent>(TEvent @event, CancellationToken cancellationToken = default)
        where TEvent : IDomainEvent;
    
    /// <summary>
    /// 发布多个事件
    /// </summary>
    Task PublishAllAsync(IEnumerable<IDomainEvent> events, 
        CancellationToken cancellationToken = default);
}

// 内存事件总线(单机版)
public class InMemoryEventBus : IEventBus
{
    private readonly IServiceProvider _serviceProvider;
    private readonly ILogger<InMemoryEventBus> _logger;
    
    public InMemoryEventBus(IServiceProvider serviceProvider, 
        ILogger<InMemoryEventBus> logger)
    {
        _serviceProvider = serviceProvider;
        _logger = logger;
    }
    
    public async Task PublishAsync<TEvent>(TEvent @event, 
        CancellationToken cancellationToken = default) where TEvent : IDomainEvent
    {
        _logger.LogInformation("Publishing event: {EventType} (Id: {EventId})", 
            @event.EventType, @event.EventId);
        
        // 获取所有该事件类型的处理器
        var handlers = _serviceProvider.GetServices<IEventHandler<TEvent>>();
        
        foreach (var handler in handlers)
        {
            try
            {
                await handler.Handle(@event, cancellationToken);
                _logger.LogDebug("Handler {HandlerType} processed event successfully", 
                    handler.GetType().Name);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Handler {HandlerType} failed to process event", 
                    handler.GetType().Name);
                
                // 根据策略决定是否继续或中断
                throw;  // 中断并抛出异常
            }
        }
    }
    
    public async Task PublishAllAsync(IEnumerable<IDomainEvent> events, 
        CancellationToken cancellationToken = default)
    {
        foreach (var @event in events)
        {
            // 使用反射调用泛型方法
            var eventType = @event.GetType();
            var method = typeof(InMemoryEventBus)
                .GetMethod(nameof(PublishAsync))
                ?.MakeGenericMethod(eventType);
            
            if (method != null)
            {
                await (Task)method.Invoke(this, new object[] { @event, cancellationToken });
            }
        }
    }
}

3. 事件发布策略 ​

3.1 同步 vs 异步发布 ​

同步发布(在同一事务内):

csharp
public class SyncEventPublisherInterceptor : SaveChangesInterceptor
{
    private readonly IEventBus _eventBus;
    
    public SyncEventPublisherInterceptor(IEventBus eventBus)
    {
        _eventBus = eventBus;
    }
    
    public override async ValueTask<int> SavedChangesAsync(
        SaveChangesCompletedEventData eventData,
        int result,
        CancellationToken cancellationToken = default)
    {
        var context = eventData.Context;
        
        if (context != null)
        {
            // 获取所有包含领域事件的实体
            var entitiesWithEvents = context.ChangeTracker.Entries<AggregateRoot>()
                .Select(e => e.Entity)
                .Where(e => e.DomainEvents.Any())
                .ToList();
            
            // 同步发布所有事件(在同一事务内)
            foreach (var entity in entitiesWithEvents)
            {
                var events = entity.DomainEvents.ToList();
                
                foreach (var @event in events)
                {
                    await _eventBus.PublishAsync(@event, cancellationToken);
                }
                
                entity.ClearDomainEvents();
            }
        }
        
        return await base.SavedChangesAsync(eventData, result, cancellationToken);
    }
}

// ✅ 优点: 强一致性,事件处理和数据库操作要么都成功,要么都失败
// ❌ 缺点: 性能较低,事件处理失败会影响主业务流程

异步发布(事务外):

csharp
public class AsyncEventPublisherInterceptor : SaveChangesInterceptor
{
    private readonly IEventBus _eventBus;
    private readonly ConcurrentQueue<IDomainEvent> _pendingEvents = new();
    
    public AsyncEventPublisherInterceptor(IEventBus eventBus)
    {
        _eventBus = eventBus;
    }
    
    public override async ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        var context = eventData.Context;
        
        if (context != null)
        {
            var entitiesWithEvents = context.ChangeTracker.Entries<AggregateRoot>()
                .Select(e => e.Entity)
                .Where(e => e.DomainEvents.Any())
                .ToList();
            
            // 收集事件到队列
            foreach (var entity in entitiesWithEvents)
            {
                foreach (var @event in entity.DomainEvents)
                {
                    _pendingEvents.Enqueue(@event);
                }
                
                entity.ClearDomainEvents();
            }
        }
        
        return await base.SavingChangesAsync(eventData, result, cancellationToken);
    }
    
    public override async ValueTask<int> SavedChangesAsync(
        SaveChangesCompletedEventData eventData,
        int result,
        CancellationToken cancellationToken = default)
    {
        // 异步发布事件(事务已提交)
        while (!_pendingEvents.IsEmpty)
        {
            if (_pendingEvents.TryDequeue(out var @event))
            {
                // Fire and forget - 不等待事件处理完成
                _ = Task.Run(async () =>
                {
                    try
                    {
                        await _eventBus.PublishAsync(@event, cancellationToken);
                    }
                    catch (Exception ex)
                    {
                        // 记录错误,但不影响主流程
                        Console.WriteLine($"Event publishing failed: {ex.Message}");
                    }
                }, cancellationToken);
            }
        }
        
        return await base.SavedChangesAsync(eventData, result, cancellationToken);
    }
}

// ✅ 优点: 高性能,不阻塞主流程
// ❌ 缺点: 最终一致性,事件可能处理失败

protected override void OnModelCreating(ModelBuilder modelBuilder)

public class OutboxInterceptor : SaveChangesInterceptor
{
    public override async ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        var context = eventData.Context;
        
        if (context != null)
        {
            var entitiesWithEvents = context.ChangeTracker.Entries<AggregateRoot>()
                .Select(e => e.Entity)
                .Where(e => e.DomainEvents.Any())
                .ToList();
            
            foreach (var entity in entitiesWithEvents)
            {
                foreach (var @event in entity.DomainEvents)
                {
                    // 将事件保存到 Outbox 表
                    var outboxMessage = new OutboxMessage
                    {
                        Id = Guid.NewGuid(),
                        EventType = @event.GetType().FullName,
                        Data = JsonSerializer.Serialize(@event),
                        OccurredOn = @event.OccurredOn
                    };
                    
                    context.Set<OutboxMessage>().Add(outboxMessage);
                }
                
                entity.ClearDomainEvents();
            }
        }
        
        return await base.SavingChangesAsync(eventData, result, cancellationToken);
    }
}

Outbox 处理器(后台任务):

csharp
public class OutboxProcessor : BackgroundService
{
    private readonly IServiceProvider _serviceProvider;
    private readonly ILogger<OutboxProcessor> _logger;
    private readonly TimeSpan _pollingInterval = TimeSpan.FromSeconds(5);
    
    public OutboxProcessor(IServiceProvider serviceProvider, 
        ILogger<OutboxProcessor> logger)
    {
        _serviceProvider = serviceProvider;
        _logger = logger;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await ProcessOutboxMessagesAsync(stoppingToken);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error processing outbox messages");
            }
            
            await Task.Delay(_pollingInterval, stoppingToken);
        }
    }
    
    private async Task ProcessOutboxMessagesAsync(CancellationToken cancellationToken)
    {
        using var scope = _serviceProvider.CreateScope();
        var context = scope.ServiceProvider.GetRequiredService<AppDbContext>();
        var eventBus = scope.ServiceProvider.GetRequiredService<IEventBus>();
        
        // 获取未处理的消息
        var messages = await context.Set<OutboxMessage>()
            .Where(m => m.ProcessedOn == null && m.RetryCount < m.MaxRetries)
            .OrderBy(m => m.OccurredOn)
            .Take(100)  // 批量处理
            .ToListAsync(cancellationToken);
        
        foreach (var message in messages)
        {
            try
            {
                // 反序列化事件
                var eventType = Type.GetType(message.EventType);
                if (eventType == null) continue;
                
                var @event = JsonSerializer.Deserialize(message.Data, eventType);
                
                // 发布事件
                await eventBus.PublishAsync((IDomainEvent)@event, cancellationToken);
                
                // 标记为已处理
                message.ProcessedOn = DateTime.UtcNow;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to process outbox message {MessageId}", 
                    message.Id);
                
                // 增加重试计数
                message.RetryCount++;
                message.ErrorMessage = ex.Message;
            }
        }
        
        await context.SaveChangesAsync(cancellationToken);
    }
}

// 注册后台服务
builder.Services.AddHostedService<OutboxProcessor>();

// 事件处理器接口
public interface IEventHandler<in TEvent> where TEvent : IDomainEvent
{
    Task Handle(TEvent @event, CancellationToken cancellationToken = default);

5.2 事件溯源实现 ​

csharp
public class EventStore
{
    private readonly AppDbContext _context;
    
    public EventStore(AppDbContext context)
    {
        _context = context;
    }
    
    // 保存事件
    public async Task SaveEventAsync(IDomainEvent @event, string aggregateId, 
        string aggregateType, CancellationToken cancellationToken = default)
    {
        var storedEvent = new StoredEvent
        {
            EventId = @event.EventId,
            EventType = @event.GetType().FullName,
            AggregateId = aggregateId,
            AggregateType = aggregateType,
            Data = JsonSerializer.Serialize(@event),
            Metadata = JsonSerializer.Serialize(new
            {
                UserId = GetCurrentUserId(),
                IpAddress = GetIpAddress(),
                Timestamp = DateTime.UtcNow
            }),
            OccurredOn = @event.OccurredOn
        };
        
        _context.Set<StoredEvent>().Add(storedEvent);
        await _context.SaveChangesAsync(cancellationToken);
    }
    
    // 获取聚合的事件流
    public async Task<IEnumerable<IDomainEvent>> GetEventsForAggregateAsync(
        string aggregateId, string aggregateType, 
        CancellationToken cancellationToken = default)
    {
        var storedEvents = await _context.Set<StoredEvent>()
            .Where(e => e.AggregateId == aggregateId && 
                       e.AggregateType == aggregateType)
            .OrderBy(e => e.OccurredOn)
            .ToListAsync(cancellationToken);
        
        return storedEvents.Select(se => DeserializeEvent(se));
    }
    
    // 重放事件重建状态
    public async Task<TAggregate> RebuildAggregateAsync<TAggregate>(
        string aggregateId, string aggregateType,
        CancellationToken cancellationToken = default) 
        where TAggregate : AggregateRoot, new()
    {
        var events = await GetEventsForAggregateAsync(
            aggregateId, aggregateType, cancellationToken);
        
        var aggregate = new TAggregate();
        
        foreach (var @event in events)
        {
            // 应用事件到聚合
            aggregate.ApplyEvent(@event);
        }
        
        return aggregate;
    }
}

6.1 RabbitMQ 集成 ​

csharp
public class RabbitMQEventBus : IEventBus, IDisposable
{
    private readonly IConnection _connection;
    private readonly IModel _channel;
    private readonly IServiceProvider _serviceProvider;
    private readonly string _exchangeName = "domain_events";
    
    public RabbitMQEventBus(IConfiguration configuration, 
        IServiceProvider serviceProvider)
    {
        var factory = new ConnectionFactory
        {
            HostName = configuration["RabbitMQ:Host"],
            UserName = configuration["RabbitMQ:Username"],
            Password = configuration["RabbitMQ:Password"]
        };
        
        _connection = factory.CreateConnection();
        _channel = _connection.CreateModel();
        _channel.ExchangeDeclare(_exchangeName, ExchangeType.Fanout, durable: true);
        
        _serviceProvider = serviceProvider;
    }
    
    public async Task PublishAsync<TEvent>(TEvent @event, 
        CancellationToken cancellationToken = default) where TEvent : IDomainEvent
    {
        var eventName = @event.GetType().Name;
        var body = JsonSerializer.Serialize(@event);
        var properties = _channel.CreateBasicProperties();
        properties.Persistent = true;
        properties.MessageId = @event.EventId.ToString();
        properties.Timestamp = new AmqpTimestamp(DateTimeOffset.UtcNow.ToUnixTimeSeconds());
        
        _channel.BasicPublish(
            exchange: _exchangeName,
            routingKey: "",
            basicProperties: properties,
            body: Encoding.UTF8.GetBytes(body));
    }
    
    public void Subscribe<TEvent, THandler>() 
        where TEvent : IDomainEvent
        where THandler : IEventHandler<TEvent>
    {
        var queueName = _channel.QueueDeclare().QueueName;
        _channel.QueueBind(queue: queueName, exchange: _exchangeName, routingKey: "");
        
        var consumer = new EventingBasicConsumer(_channel);
        consumer.Received += async (model, ea) =>
        {
            var body = ea.Body.ToArray();
            var message = Encoding.UTF8.GetString(body);
            var @event = JsonSerializer.Deserialize<TEvent>(message);
            
            using var scope = _serviceProvider.CreateScope();
            var handler = scope.ServiceProvider.GetRequiredService<THandler>();
            
            await handler.Handle(@event);
            
            _channel.BasicAck(deliveryTag: ea.DeliveryTag, multiple: false);
        };
        
        _channel.BasicConsume(queue: queueName, autoAck: false, consumer: consumer);
    }
    
    public void Dispose()
    {
        _channel?.Close();
        _connection?.Close();
    }
}

7.1 事务边界管理 ​

csharp
public class OrderService
{
    private readonly AppDbContext _context;
    private readonly IEventBus _eventBus;
    
    public async Task CreateOrderAsync(CreateOrderRequest request)
    {
        // 开启事务
        using var transaction = await _context.Database.BeginTransactionAsync();
        
        try
        {
            // 1. 创建订单
            var order = Order.Create(request.CustomerId, request.Items, request.ShippingAddress);
            _context.Orders.Add(order);
            await _context.SaveChangesAsync();
            
            // 2. 扣减库存
            await DeductInventory(request.Items);
            
            // 3. 创建支付记录
            var payment = Payment.Create(order.Id, request.PaymentInfo);
            _context.Payments.Add(payment);
            await _context.SaveChangesAsync();
            
            // 4. 提交事务
            await transaction.CommitAsync();
            
            // 5. 发布事件(事务外)
            foreach (var @event in order.DomainEvents)
            {
                await _eventBus.PublishAsync(@event);
            }
        }
        catch
        {
            await transaction.RollbackAsync();
            throw;
        }
    }
}

csharp
public class MonitoredEventBus : IEventBus
{
    private readonly IEventBus _inner;
    private readonly IMetricsCollector _metrics;
    
    public async Task PublishAsync<TEvent>(TEvent @event, 
        CancellationToken cancellationToken = default)
    {
        var stopwatch = Stopwatch.StartNew();
        
        try
        {
            await _inner.PublishAsync(@event, cancellationToken);
            
            _metrics.RecordSuccess("event_published", @event.EventType);
        }
        catch (Exception ex)
        {
            _metrics.RecordFailure("event_publish_failed", @event.EventType);
            throw;
        }
        finally
        {
            stopwatch.Stop();
            _metrics.RecordDuration("event_publish_duration", stopwatch.ElapsedMilliseconds);
        }
    }
}

总结 ​

领域事件是实现解耦和可扩展性的强大模式:

核心价值 ​

✅ 解耦业务逻辑 - 发布者和订阅者独立演化
✅ 支持复杂工作流 - 轻松添加新的事件处理器
✅ 审计与追溯 - 完整的事件历史记录
✅ 分布式系统集成 - 天然适合微服务架构

关键模式 ​

  1. 同步发布 - 强一致性,简单场景
  2. 异步发布 - 高性能,最终一致性
  3. Outbox - 可靠消息传递
  4. 事件溯源 - 完整的历史追踪

掌握领域事件,你可以构建出高度解耦和可扩展的企业级应用!
