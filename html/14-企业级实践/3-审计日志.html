审计日志(Audit Logging) ​

审计日志是企业级应用的核心功能,用于追踪系统中发生的所有重要操作。本章将深入探讨如何在 EF Core 应用中实现完整的审计日志系统,包括自动审计、数据变更追踪、用户行为分析以及合规性报告。

1. 审计日志基础 ​

1.1 为什么需要审计日志? ​

核心价值:

csharp
// 业务价值:
✅ 安全监控 - 检测异常行为和未授权访问
✅ 问题诊断 - 追溯数据变更历史,快速定位bug
✅ 合规要求 - GDPR, SOX, HIPAA等法规强制要求
✅ 责任追溯 - 明确谁在什么时候做了什么
✅ 数据分析 - 用户行为分析和业务流程优化

// 实际场景:
- "谁删除了这个订单?" 
- "这个价格是什么时候修改的?"
- "哪些用户在非工作时间访问了敏感数据?"
- "恢复误删的客户记录"
- "生成月度操作审计报告"

csharp
// 需要审计的操作:
1. 数据创建(CREATE) - 谁创建了记录
2. 数据更新(UPDATE) - 修改了哪些字段,旧值和新值
3. 数据删除(DELETE) - 删除了什么记录

- When: 操作时间戳(UTC)
- What: 实体类型、实体ID、操作类型
- How: 使用的API端点、User-Agent
- Why: 操作原因(可选,业务备注)
- Changes: 具体变更内容(旧值→新值)

    public string? NewValues { get; set; }  // JSON

    public string UserName { get; set; } = null!;
    public string? IpAddress { get; set; }
    public string? UserAgent { get; set; }
    
    // 上下文信息
    public string? Endpoint { get; set; }
    public string? ActionMethod { get; set; }
    
    // 时间戳
    public DateTime Timestamp { get; set; } = DateTime.UtcNow;
    
    // 执行信息
    public TimeSpan? ExecutionTime { get; set; }  // 操作耗时
    public bool Success { get; set; } = true;
    public string? ErrorMessage { get; set; }
}

审计拦截器核心实现:

csharp
public class AuditInterceptor : SaveChangesInterceptor
{
    private readonly IAuditService _auditService;
    private readonly List<AuditEntry> _auditEntries = new();
    
    public AuditInterceptor(IAuditService auditService)
    {
        _auditService = auditService;
    }
    
    public override async ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        var context = eventData.Context;
        if (context == null) return result;
        
        // 收集审计信息
        foreach (var entry in context.ChangeTracker.Entries())
        {
            // 跳过审计记录本身,避免无限循环
            if (entry.Entity is AuditRecord) continue;
            
            // 只审计实现了 IAuditable 的实体
            if (entry.Entity is not IAuditable) continue;
            
            var auditEntry = CreateAuditEntry(entry);
            if (auditEntry != null)
            {
                _auditEntries.Add(auditEntry);
            }
        }
        
        return await base.SavingChangesAsync(eventData, result, cancellationToken);
    }
    
    public override async ValueTask<int> SavedChangesAsync(
        SaveChangesCompletedEventData eventData,
        int result,
        CancellationToken cancellationToken = default)
    {
        var context = eventData.Context;
        
        // 保存成功后,写入审计日志
        if (context != null && _auditEntries.Any())
        {
            try
            {
                await _auditService.LogAsync(_auditEntries, cancellationToken);
            }
            catch (Exception ex)
            {
                // 审计失败不应影响主业务
                Console.WriteLine($"Audit logging failed: {ex.Message}");
            }
            finally
            {
                _auditEntries.Clear();
            }
        }
        
        return await base.SavedChangesAsync(eventData, result, cancellationToken);
    }
    
    private AuditEntry? CreateAuditEntry(EntityEntry entry)
    {
        return entry.State switch
        {
            EntityState.Added => CreateAddedAuditEntry(entry),
            EntityState.Modified => CreateModifiedAuditEntry(entry),
            EntityState.Deleted => CreateDeletedAuditEntry(entry),
            _ => null
        };
    }
    
    private AuditEntry CreateAddedAuditEntry(EntityEntry entry)
    {
        var audit = new AuditEntry
        {
            EntityName = entry.Entity.GetType().Name,
            EntityId = GetEntityId(entry),
            ActionType = "Create",
            NewValues = SerializeProperties(entry.Properties.ToDictionary(
                p => p.Metadata.Name,
                p => p.CurrentValue))
        };
        
        return audit;
    }
    
    private AuditEntry? CreateModifiedAuditEntry(EntityEntry entry)
    {
        var changedProps = entry.Properties
            .Where(p => p.IsModified)
            .ToList();
        
        if (!changedProps.Any()) return null;
        
        var audit = new AuditEntry
        {
            EntityName = entry.Entity.GetType().Name,
            EntityId = GetEntityId(entry),
            ActionType = "Update",
            OldValues = SerializeProperties(changedProps.ToDictionary(
                p => p.Metadata.Name,
                p => p.OriginalValue)),
            NewValues = SerializeProperties(changedProps.ToDictionary(
                p => p.Metadata.Name,
                p => p.CurrentValue)),
            ChangedProperties = JsonSerializer.Serialize(
                changedProps.Select(p => p.Metadata.Name))
        };
        
        return audit;
    }
    
    private AuditEntry CreateDeletedAuditEntry(EntityEntry entry)
    {
        var audit = new AuditEntry
        {
            EntityName = entry.Entity.GetType().Name,
            EntityId = GetEntityId(entry),
            ActionType = "Delete",
            OldValues = SerializeProperties(entry.Properties.ToDictionary(
                p => p.Metadata.Name,
                p => p.OriginalValue))
        };
        
        return audit;
    }
    
    private string GetEntityId(EntityEntry entry)
    {
        var keyProperties = entry.Metadata.FindPrimaryKey()?.Properties;
        if (keyProperties != null && keyProperties.Any())
        {
            var keyValues = keyProperties.Select(kp => 
                entry.Property(kp.Name).CurrentValue);
            return string.Join(",", keyValues);
        }
        
        return "Unknown";
    }
    
    private string SerializeProperties(Dictionary<string, object?> properties)
    {
        // 排除敏感字段
        var sanitized = properties.Where(kvp => 
            !kvp.Key.Contains("Password", StringComparison.OrdinalIgnoreCase) &&
            !kvp.Key.Contains("Secret", StringComparison.OrdinalIgnoreCase) &&
            !kvp.Key.Contains("Token", StringComparison.OrdinalIgnoreCase))
            .ToDictionary(kvp => kvp.Key, kvp => kvp.Value);
        
        return JsonSerializer.Serialize(sanitized, new JsonSerializerOptions
        {
            WriteIndented = false,
            Encoder = System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping
        });
    }
}

2.2 审计服务实现 ​

csharp
public interface IAuditService
{
    Task LogAsync(IEnumerable<AuditEntry> entries, CancellationToken ct = default);
    Task<AuditRecord> GetByIdAsync(long id, CancellationToken ct = default);
    Task<List<AuditRecord>> QueryAsync(AuditQuery query, CancellationToken ct = default);
}

public class AuditService : IAuditService
{
    private readonly AppDbContext _context;
    private readonly IUserContext _userContext;
    private readonly IHttpContextAccessor _httpContextAccessor;
    
    public AuditService(
        AppDbContext context,
        IUserContext userContext,
        IHttpContextAccessor httpContextAccessor)
    {
        _context = context;
        _userContext = userContext;
        _httpContextAccessor = httpContextAccessor;
    }
    
    public async Task LogAsync(IEnumerable<AuditEntry> entries, CancellationToken ct = default)
    {
        var auditRecords = entries.Select(entry => new AuditRecord
        {
            EntityName = entry.EntityName,
            EntityId = entry.EntityId,
            ActionType = entry.ActionType,
            OldValues = entry.OldValues,
            NewValues = entry.NewValues,
            ChangedProperties = entry.ChangedProperties,
            
            // 用户信息
            UserId = _userContext.UserId ?? "System",
            UserName = _userContext.UserName ?? "System",
            IpAddress = GetIpAddress(),
            UserAgent = GetUserAgent(),
            
            // 上下文
            Endpoint = GetEndpoint(),
            ActionMethod = GetActionMethod(),
            
            Timestamp = DateTime.UtcNow,
            Success = true
        }).ToList();
        
        _context.AuditRecords.AddRange(auditRecords);
        await _context.SaveChangesAsync(ct);
    }
    
    private string? GetIpAddress()
    {
        return _httpContextAccessor.HttpContext?.Connection.RemoteIpAddress?.ToString();
    }
    
    private string? GetUserAgent()
    {
        return _httpContextAccessor.HttpContext?.Request.Headers["User-Agent"].FirstOrDefault();
    }
    
    private string? GetEndpoint()
    {
        return _httpContextAccessor.HttpContext?.Request.Path.ToString();
    }
    
    private string? GetActionMethod()
    {
        return _httpContextAccessor.HttpContext?.Request.Method;
    }
    
    public async Task<AuditRecord> GetByIdAsync(long id, CancellationToken ct = default)
    {
        return await _context.AuditRecords
            .AsNoTracking()
            .FirstOrDefaultAsync(a => a.Id == id, ct);
    }
    
    public async Task<List<AuditRecord>> QueryAsync(AuditQuery query, CancellationToken ct = default)
    {
        var qry = _context.AuditRecords.AsNoTracking();
        
        if (!string.IsNullOrEmpty(query.EntityName))
            qry = qry.Where(a => a.EntityName == query.EntityName);
        
        if (!string.IsNullOrEmpty(query.EntityId))
            qry = qry.Where(a => a.EntityId == query.EntityId);
        
        if (!string.IsNullOrEmpty(query.UserId))
            qry = qry.Where(a => a.UserId == query.UserId);
        
        if (!string.IsNullOrEmpty(query.ActionType))
            qry = qry.Where(a => a.ActionType == query.ActionType);
        
        if (query.StartDate.HasValue)
            qry = qry.Where(a => a.Timestamp >= query.StartDate);
        
        if (query.EndDate.HasValue)
            qry = qry.Where(a => a.Timestamp <= query.EndDate);
        
        return await qry
            .OrderByDescending(a => a.Timestamp)
            .Skip((query.Page - 1) * query.PageSize)
            .Take(query.PageSize)
            .ToListAsync(ct);
    }
}

3. 数据变更追踪 ​

3.1 详细的变更对比 ​

csharp
public class ChangeTracker
{
    /// <summary>
    /// 生成人类可读的变更描述
    /// </summary>
    public static string GenerateChangeDescription(AuditRecord audit)
    {
        if (audit.ActionType == "Create")
        {
            return $"Created new {audit.EntityName} (ID: {audit.EntityId})";
        }
        
        if (audit.ActionType == "Delete")
        {
            return $"Deleted {audit.EntityName} (ID: {audit.EntityId})";
        }
        
        if (audit.ActionType == "Update" && !string.IsNullOrEmpty(audit.ChangedProperties))
        {
            var properties = JsonSerializer.Deserialize<List<string>>(audit.ChangedProperties);
            var oldValues = JsonSerializer.Deserialize<Dictionary<string, object>>(audit.OldValues!);
            var newValues = JsonSerializer.Deserialize<Dictionary<string, object>>(audit.NewValues!);
            
            var changes = new List<string>();
            foreach (var prop in properties!)
            {
                var oldVal = oldValues.ContainsKey(prop) ? oldValues[prop] : "null";
                var newVal = newValues.ContainsKey(prop) ? newValues[prop] : "null";
                changes.Add($"{prop}: {oldVal} → {newVal}");
            }
            
            return $"Updated {audit.EntityName} (ID: {audit.EntityId}): {string.Join(", ", changes)}";
        }
        
        return $"{audit.ActionType} {audit.EntityName} (ID: {audit.EntityId})";
    }
}

// 使用示例
var audit = await context.AuditRecords.FindAsync(123);
var description = ChangeTracker.GenerateChangeDescription(audit);
Console.WriteLine(description);
// 输出: "Updated Product (ID: 456): Price: 99.99 → 149.99, Stock: 100 → 85"

csharp
public class AuditHistoryDto
{
    public long AuditId { get; set; }
    public string Action { get; set; }
    public string UserName { get; set; }
    public DateTime Timestamp { get; set; }
    public List<PropertyChange> Changes { get; set; }
}

public class PropertyChange
{
    public string PropertyName { get; set; }
    public object? OldValue { get; set; }
    public object? NewValue { get; set; }
}

// API端点: 获取实体的审计历史
app.Get("/api/audit/{entityName}/{entityId}/history", async (
    string entityName,
    string entityId,
    AppDbContext context) =>
{
    var audits = await context.AuditRecords
        .Where(a => a.EntityName == entityName && a.EntityId == entityId)
        .OrderByDescending(a => a.Timestamp)
        .Take(50)
        .ToListAsync();
    
    var history = audits.Select(a => new AuditHistoryDto
    {
        AuditId = a.Id,
        Action = a.ActionType,
        UserName = a.UserName,
        Timestamp = a.Timestamp,
        Changes = ParseChanges(a)
    }).ToList();
    
    return Results.Ok(history);
});

List<PropertyChange> ParseChanges(AuditRecord audit)
{
    if (string.IsNullOrEmpty(audit.ChangedProperties))
        return new List<PropertyChange>();
    
    var properties = JsonSerializer.Deserialize<List<string>>(audit.ChangedProperties);
    var oldValues = JsonSerializer.Deserialize<Dictionary<string, object>>(audit.OldValues!);
    var newValues = JsonSerializer.Deserialize<Dictionary<string, object>>(audit.NewValues!);
    
    return properties!.Select(prop => new PropertyChange
    {
        PropertyName = prop,
        OldValue = oldValues.ContainsKey(prop) ? oldValues[prop] : null,
        NewValue = newValues.ContainsKey(prop) ? newValues[prop] : null
    }).ToList();
}

    public Dictionary<string, string> Claims => 
        _httpContextAccessor.HttpContext?.User.Claims
            .ToDictionary(c => c.Type, c => c.Value) 
            ?? new Dictionary<string, string>();
}

    if (sensitiveAttr != null)
    {
        var auditService = context.RequestServices.GetRequiredService<IAuditService>();

            userName: userContext.UserName,
            reason: sensitiveAttr.Reason,
            ipAddress: context.Connection.RemoteIpAddress?.ToString(),
            timestamp: DateTime.UtcNow);
    }
    
    await next(context);
});

public class AuditArchivalService
{
    private readonly AppDbContext _context;
    private readonly ILogger<AuditArchivalService> _logger;
    
    /// <summary>
    /// 归档旧审计数据到冷存储
    /// </summary>
    public async Task ArchiveOldAuditsAsync(DateTime beforeDate, string archivePath)
    {
        var cutoffDate = beforeDate.AddMonths(-6);  // 归档6个月前的数据
        
        // 1. 导出到JSON文件
        var oldAudits = await _context.AuditRecords
            .Where(a => a.Timestamp < cutoffDate)
            .OrderBy(a => a.Timestamp)
            .ToListAsync();
        
        var json = JsonSerializer.Serialize(oldAudits, new JsonSerializerOptions 
        { 
            WriteIndented = true 
        });
        
        var fileName = $"audit_archive_{cutoffDate:yyyy-MM}.json.gz";
        var filePath = Path.Combine(archivePath, fileName);
        
        await CompressAndSaveAsync(json, filePath);
        
        // 2. 从主表删除
        var deletedCount = await _context.AuditRecords
            .Where(a => a.Timestamp < cutoffDate)
            .ExecuteDeleteAsync();
        
        _logger.LogInformation("Archived {Count} audit records to {File}", 
            deletedCount, filePath);
    }
    
    private async Task CompressAndSaveAsync(string json, string filePath)
    {
        using var fileStream = File.Create(filePath);
        using var gzipStream = new GZipStream(fileStream, CompressionLevel.Optimal);
        using var writer = new StreamWriter(gzipStream);
        await writer.WriteAsync(json);
    }
}

public class AuditAnalytics
{
    private readonly AppDbContext _context;
    
    /// <summary>
    /// 用户活动统计
    /// </summary>
    public async Task<UserActivityStats> GetUserActivityAsync(string userId, DateTime fromDate)
    {
        var audits = await _context.AuditRecords
            .Where(a => a.UserId == userId && a.Timestamp >= fromDate)
            .GroupBy(a => a.ActionType)
            .Select(g => new { ActionType = g.Key, Count = g.Count() })
            .ToListAsync();
        
        return new UserActivityStats
        {
            UserId = userId,
            Creates = audits.FirstOrDefault(x => x.ActionType == "Create")?.Count ?? 0,
            Updates = audits.FirstOrDefault(x => x.ActionType == "Update")?.Count ?? 0,
            Deletes = audits.FirstOrDefault(x => x.ActionType == "Delete")?.Count ?? 0,
            TotalActions = audits.Sum(x => x.Count)
        };
    }
    
    /// <summary>
    /// 热点数据分析(最常被修改的实体)
    /// </summary>
    public async Task<List<HotEntityStats>> GetHotEntitiesAsync(int topN = 10)
    {
        return await _context.AuditRecords
            .Where(a => a.Timestamp >= DateTime.UtcNow.AddDays(-30))
            .GroupBy(a => new { a.EntityName, a.EntityId })
            .Select(g => new HotEntityStats
            {
                EntityName = g.Key.EntityName,
                EntityId = g.Key.EntityId,
                UpdateCount = g.Count(a => a.ActionType == "Update"),
                LastModified = g.Max(a => a.Timestamp),
                ModifiedBy = g.OrderByDescending(a => a.Timestamp)
                    .Select(a => a.UserName)
                    .FirstOrDefault()
            })
            .OrderByDescending(x => x.UpdateCount)
            .Take(topN)
            .ToListAsync();
    }
    
    /// <summary>
    /// 异常行为检测(非工作时间操作)
    /// </summary>
    public async Task<List<AuditRecord>> GetSuspiciousActivitiesAsync()
    {
        return await _context.AuditRecords
            .Where(a => a.Timestamp.Hour < 6 || a.Timestamp.Hour > 22)  // 非工作时间
            .Where(a => a.ActionType == "Delete" || a.ActionType == "Update")
            .Where(a => a.Timestamp >= DateTime.UtcNow.AddDays(-7))
            .OrderByDescending(a => a.Timestamp)
            .Take(50)
            .ToListAsync();
    }
}

    /// 生成GDPR数据访问报告

        var audits = await _context.AuditRecords
            .Where(a => a.UserId == userId)
            .Where(a => a.Timestamp >= periodStart && a.Timestamp <= periodEnd)
            .OrderByDescending(a => a.Timestamp)
            .ToListAsync();
        
        return new GdprReport
        {
            UserId = userId,
            PeriodStart = periodStart,
            PeriodEnd = periodEnd,

                .ToDictionary(g => g.Key, g => g.Count()),
            AccessedEntities = audits.Select(a => new AccessedEntity
            {
                EntityType = a.EntityName,
                EntityId = a.EntityId,
                Action = a.ActionType,
                Timestamp = a.Timestamp
            }).ToList(),
            GeneratedAt = DateTime.UtcNow
        };
    }

            ReportFormat.Excel => GenerateExcelReport(audits),
            ReportFormat.CSV => GenerateCsvReport(audits),
            _ => throw new ArgumentException("Unsupported format")
        };
    }

csharp
public class AuditPerformanceMonitor
{
    private readonly IMetricsCollector _metrics;
    
    public async Task MonitorAuditPerformanceAsync(Func<Task> auditAction)
    {
        var stopwatch = Stopwatch.StartNew();
        
        try
        {
            await auditAction();
            
            _metrics.RecordSuccess("audit_write");
        }
        catch (Exception ex)
        {
            _metrics.RecordFailure("audit_write_failed");
            throw;
        }
        finally
        {
            stopwatch.Stop();
            _metrics.RecordDuration("audit_duration_ms", stopwatch.ElapsedMilliseconds);
            
            // 告警: 审计超过1秒
            if (stopwatch.ElapsedMilliseconds > 1000)
            {
                _logger.LogWarning("Audit logging took {Ms}ms", stopwatch.ElapsedMilliseconds);
            }
        }
    }
}
