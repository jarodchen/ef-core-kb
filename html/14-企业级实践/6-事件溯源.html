事件溯源(Event Sourcing) ​

事件溯源是一种架构模式,通过存储状态变更的事件序列而非当前状态本身来持久化实体状态。本章将深入探讨如何在 EF Core 应用中实现事件溯源,包括事件存储、快照优化、时间旅行调试以及与传统CRUD的对比。

1. 事件溯源基础 ​

1.1 什么是事件溯源? ​

核心概念对比:

csharp
// ===== 传统CRUD - 存储当前状态 =====
// 数据库只保存最新状态
Products Table:
| Id | Name     | Price | Stock |
|----|----------|-------|-------|
| 1  | Laptop   | 999   | 85    |  ← 只知道当前状态

// 问题: 
// ❌ 无法追溯历史变更
// ❌ 不知道价格是如何变化的
// ❌ 无法重现某个时间点的状态

// ===== 事件溯源 - 存储事件流 =====
// 数据库保存所有事件
Events Table:
| AggregateId | EventType           | Data                    | Timestamp           |
|-------------|---------------------|-------------------------|---------------------|
| 1           | ProductCreated      | {Name:"Laptop",...}    | 2024-01-01 10:00   |
| 1           | PriceChanged        | {Old:899, New:999}     | 2024-01-15 14:30   |
| 1           | StockAdjusted       | {Change:-15}           | 2024-02-01 09:15   |
| 1           | StockAdjusted       | {Change:-5}            | 2024-02-10 16:45   |

// 优势:
// ✅ 完整的历史记录
// ✅ 可以重放事件重建任意时刻的状态
// ✅ 天然的审计日志
// ✅ 支持时间旅行调试

事件溯源的优势:

✅ 完整审计 - 每个变更都有记录
✅ 时间旅行 - 重现历史状态
✅ 调试友好 - 追溯bug根源
✅ 业务洞察 - 分析行为模式
✅ 合规性 - 满足严格的数据保留要求

适用场景:

✅ 推荐事件溯源:

  • 金融系统(需要完整审计)
  • 订单处理(追踪状态流转)
  • 库存管理(精确追溯)
  • 医疗记录(法规要求)
  • 协作系统(冲突解决)

❌ 不推荐事件溯源:

  • 简单CRUD应用
  • 不需要历史记录
  • 性能要求极高
  • 团队经验不足

1.2 事件溯源 vs 传统CRUD ​

csharp
// ===== 传统CRUD流程 =====
// 1. 从数据库加载当前状态
var product = await context.Products.FindAsync(1);

// 2. 修改状态
product.Price = 999;
product.Stock -= 5;

// 3. 保存新状态(覆盖旧状态)
await context.SaveChangesAsync();

// 结果: 旧状态丢失!

// ===== 事件溯源流程 =====
// 1. 从事件存储加载事件流
var events = await eventStore.GetEventsAsync(productId);

// 2. 重放事件重建状态
var product = Product.ReplayFromHistory(events);

// 3. 执行业务逻辑,生成新事件
product.ChangePrice(999);  // 生成 PriceChanged 事件
product.AdjustStock(-5);   // 生成 StockAdjusted 事件

// 4. 追加新事件到事件流(不覆盖)
await eventStore.AppendEventsAsync(productId, product.GetUncommittedEvents());

// 结果: 完整保留所有历史!

// 配置
public class StoredEventConfiguration : IEntityTypeConfiguration<StoredEvent>
{
    public void Configure(EntityTypeBuilder<StoredEvent> builder)
    {
        builder.HasKey(e => e.Id);
        
        builder.Property(e => e.AggregateId).IsRequired();
        builder.Property(e => e.EventType).IsRequired().HasMaxLength(200);
        builder.Property(e => e.EventData).IsRequired();
        
        // 复合索引: 按聚合ID和版本查询
        builder.HasIndex(e => new { e.AggregateId, e.Version }).IsUnique();
        
        // 索引: 按聚合类型查询
        builder.HasIndex(e => e.AggregateType);
        
        // 索引: 按时间查询
        builder.HasIndex(e => e.OccurredOn);
    }
}

3. 聚合根与事件 ​

3.1 事件溯源聚合根 ​

csharp
public abstract class EventSourcedAggregate
{
    private readonly List<DomainEvent> _uncommittedEvents = new();
    
    public Guid Id { get; protected set; }
    public int Version { get; protected set; }  // 当前版本号
    
    public IReadOnlyCollection<DomainEvent> UncommittedEvents => 
        _uncommittedEvents.AsReadOnly();
    
    /// <summary>
    /// 加载事件历史,重建状态
    /// </summary>
    public void LoadFromHistory(IEnumerable<DomainEvent> history)
    {
        foreach (var @event in history)
        {
            ApplyEvent(@event);
            Version++;
        }
    }
    
    /// <summary>
    /// 添加新事件
    /// </summary>
    protected void AddEvent(DomainEvent @event)
    {
        ApplyEvent(@event);
        _uncommittedEvents.Add(@event);
        Version++;
    }
    
    /// <summary>
    /// 应用事件到状态
    /// </summary>
    protected abstract void ApplyEvent(DomainEvent @event);
    
    /// <summary>
    /// 标记事件为已提交
    /// </summary>
    public void MarkEventsAsCommitted()
    {
        _uncommittedEvents.Clear();
    }
}

// 产品聚合根
public class Product : EventSourcedAggregate
{
    public string Name { get; private set; } = null!;
    public decimal Price { get; private set; }
    public int Stock { get; private set; }
    public string Category { get; private set; } = null!;
    public bool IsActive { get; private set; }
    
    // 私有构造函数 - 强制使用工厂方法
    private Product() { }
    
    /// <summary>
    /// 创建新产品
    /// </summary>
    public static Product Create(string name, decimal price, int stock, string category)
    {
        if (string.IsNullOrWhiteSpace(name))
            throw new ArgumentException("Name is required");
        
        if (price < 0)
            throw new ArgumentException("Price cannot be negative");
        
        if (stock < 0)
            throw new ArgumentException("Stock cannot be negative");
        
        var product = new Product();
        
        // 添加创建事件
        product.AddEvent(new ProductCreatedEvent
        {
            ProductId = 0,  // 将在保存时分配
            Name = name,
            Price = price,
            Stock = stock,
            Category = category
        });
        
        return product;
    }
    
    /// <summary>
    /// 修改价格
    /// </summary>
    public void ChangePrice(decimal newPrice, string reason)
    {
        if (newPrice < 0)
            throw new ArgumentException("Price cannot be negative");
        
        if (Math.Abs(newPrice - Price) < 0.01m)
            return;  // 价格未变化
        
        AddEvent(new PriceChangedEvent
        {
            ProductId = (int)Id,
            OldPrice = Price,
            NewPrice = newPrice,
            Reason = reason
        });
    }
    
    /// <summary>
    /// 调整库存
    /// </summary>
    public void AdjustStock(int change, string reason)
    {
        var newStock = Stock + change;
        
        if (newStock < 0)
            throw new InvalidOperationException($"Insufficient stock. Current: {Stock}, Requested change: {change}");
        
        AddEvent(new StockAdjustedEvent
        {
            ProductId = (int)Id,
            Change = change,
            Reason = reason,
            RemainingStock = newStock
        });
    }
    
    /// <summary>
    /// 重命名产品
    /// </summary>
    public void Rename(string newName)
    {
        if (string.IsNullOrWhiteSpace(newName))
            throw new ArgumentException("Name is required");
        
        if (newName == Name)
            return;
        
        AddEvent(new ProductRenamedEvent
        {
            ProductId = (int)Id,
            OldName = Name,
            NewName = newName
        });
    }
    
    /// <summary>
    /// 停用产品
    /// </summary>
    public void Deactivate(string reason)
    {
        if (!IsActive)
            throw new InvalidOperationException("Product already deactivated");
        
        AddEvent(new ProductDeactivatedEvent
        {
            ProductId = (int)Id,
            Reason = reason
        });
    }
    
    /// <summary>
    /// 应用事件到状态
    /// </summary>
    protected override void ApplyEvent(DomainEvent @event)
    {
        switch (@event)
        {
            case ProductCreatedEvent e:
                Id = Guid.NewGuid();  // 或使用数据库生成的ID
                Name = e.Name;
                Price = e.Price;
                Stock = e.Stock;
                Category = e.Category;
                IsActive = true;
                break;
            
            case PriceChangedEvent e:
                Price = e.NewPrice;
                break;
            
            case StockAdjustedEvent e:
                Stock = e.RemainingStock;
                break;
            
            case ProductRenamedEvent e:
                Name = e.NewName;
                break;
            
            case ProductDeactivatedEvent e:
                IsActive = false;
                break;
        }
    }
}

4.2 EF Core事件存储实现 ​

csharp
public class EfCoreEventStore : IEventStore
{
    private readonly EventStoreDbContext _context;
    private readonly IUserContext _userContext;
    
    public EfCoreEventStore(EventStoreDbContext context, IUserContext userContext)
    {
        _context = context;
        _userContext = userContext;
    }
    
    public async Task<List<DomainEvent>> GetEventsAsync(Guid aggregateId, 
        CancellationToken ct = default)
    {
        var storedEvents = await _context.StoredEvents
            .Where(e => e.AggregateId == aggregateId)
            .OrderBy(e => e.Version)
            .ToListAsync(ct);
        
        return storedEvents.Select(DeserializeEvent).ToList();
    }
    
    public async Task<List<DomainEvent>> GetEventsAsync(Guid aggregateId, 
        int fromVersion, int toVersion, CancellationToken ct = default)
    {
        var storedEvents = await _context.StoredEvents
            .Where(e => e.AggregateId == aggregateId)
            .Where(e => e.Version >= fromVersion && e.Version <= toVersion)
            .OrderBy(e => e.Version)
            .ToListAsync(ct);
        
        return storedEvents.Select(DeserializeEvent).ToList();
    }
    
    public async Task AppendEventsAsync(Guid aggregateId, IEnumerable<DomainEvent> events, 
        int expectedVersion, CancellationToken ct = default)
    {
        // 乐观并发控制
        var currentVersion = await GetCurrentVersionAsync(aggregateId, ct);
        
        if (currentVersion != expectedVersion)
        {
            throw new ConcurrencyException(
                $"Expected version {expectedVersion}, but current version is {currentVersion}");
        }
        
        var storedEvents = new List<StoredEvent>();
        var version = expectedVersion;
        
        foreach (var @event in events)
        {
            version++;
            
            storedEvents.Add(new StoredEvent
            {
                AggregateId = aggregateId,
                AggregateType = @event.GetType().BaseType?.Name ?? "Unknown",
                Version = version,
                EventType = @event.EventType,
                EventData = JsonSerializer.Serialize(@event),
                Metadata = JsonSerializer.Serialize(new
                {
                    UserId = _userContext.UserId,
                    UserName = _userContext.UserName,
                    IpAddress = _userContext.IpAddress,
                    Timestamp = DateTime.UtcNow
                }),
                OccurredOn = @event.OccurredOn,
                CorrelationId = GetCorrelationId()
            });
        }
        
        _context.StoredEvents.AddRange(storedEvents);
        await _context.SaveChangesAsync(ct);
    }
    
    public async Task<int> GetCurrentVersionAsync(Guid aggregateId, CancellationToken ct = default)
    {
        var maxVersion = await _context.StoredEvents
            .Where(e => e.AggregateId == aggregateId)
            .MaxAsync(e => (int?)e.Version, ct);
        
        return maxVersion ?? 0;
    }
    
    private DomainEvent DeserializeEvent(StoredEvent storedEvent)
    {
        var eventType = Type.GetType($"MyApp.Domain.Events.{storedEvent.EventType}")
            ?? throw new InvalidOperationException($"Unknown event type: {storedEvent.EventType}");
        
        return (DomainEvent)JsonSerializer.Deserialize(storedEvent.EventData, eventType)!;
    }
    
    private string? GetCorrelationId()
    {
        // 从HttpContext或ExecutionContext获取
        return Activity.Current?.TraceId.ToString();
    }
}

4.3 仓库模式封装 ​

csharp
public interface IProductRepository
{
    Task<Product> GetByIdAsync(Guid id, CancellationToken ct = default);
    Task SaveAsync(Product product, CancellationToken ct = default);
}

public class ProductRepository : IProductRepository
{
    private readonly IEventStore _eventStore;
    
    public async Task<Product> GetByIdAsync(Guid id, CancellationToken ct = default)
    {
        // 从事件存储加载事件
        var events = await _eventStore.GetEventsAsync(id, ct);
        
        if (!events.Any())
            throw new NotFoundException($"Product {id} not found");
        
        // 重放事件重建状态
        var product = new Product();
        product.LoadFromHistory(events);
        
        return product;
    }
    
    public async Task SaveAsync(Product product, CancellationToken ct = default)
    {
        var uncommittedEvents = product.UncommittedEvents;
        
        if (!uncommittedEvents.Any())
            return;  // 没有变更
        
        // 追加事件到事件存储
        await _eventStore.AppendEventsAsync(
            product.Id,
            uncommittedEvents,
            product.Version - uncommittedEvents.Count,  // 期望的版本号
            ct);
        
        // 标记事件为已提交
        product.MarkEventsAsCommitted();
    }
}

5.2 快照实现 ​

csharp
public class AggregateSnapshot
{
    public Guid Id { get; set; }
    public Guid AggregateId { get; set; }
    public string AggregateType { get; set; } = null!;
    public int Version { get; set; }  // 快照时的版本
    public string SnapshotData { get; set; } = null!;  // 序列化的状态
    public DateTime CreatedAt { get; set; } = DateTime.UtcNow;
}

public class SnapshotStrategy
{
    private readonly int _snapshotThreshold;
    
    public SnapshotStrategy(int snapshotThreshold = 100)
    {
        _snapshotThreshold = snapshotThreshold;
    }
    
    public bool ShouldCreateSnapshot(int currentVersion, int lastSnapshotVersion)
    {
        return (currentVersion - lastSnapshotVersion) >= _snapshotThreshold;
    }
}

// 带快照的事件存储
public class EventStoreWithSnapshots : IEventStore
{
    private readonly IEventStore _innerStore;
    private readonly ISnapshotStore _snapshotStore;
    private readonly SnapshotStrategy _strategy;
    
    public async Task<Product> GetByIdAsync(Guid id, CancellationToken ct = default)
    {
        // 1. 尝试加载快照
        var snapshot = await _snapshotStore.GetLatestSnapshotAsync(id, ct);
        
        Product product;
        int fromVersion;
        
        if (snapshot != null)
        {
            // 从快照恢复状态
            product = DeserializeSnapshot(snapshot);
            fromVersion = snapshot.Version + 1;
        }
        else
        {
            // 从头开始
            product = new Product();
            fromVersion = 1;
        }
        
        // 2. 加载快照之后的事件
        var events = await _innerStore.GetEventsAsync(id, fromVersion, int.MaxValue, ct);
        
        // 3. 重放事件
        product.LoadFromHistory(events);
        
        return product;
    }
    
    public async Task SaveAsync(Product product, CancellationToken ct = default)
    {
        await _innerStore.AppendEventsAsync(
            product.Id,
            product.UncommittedEvents,
            product.Version - product.UncommittedEvents.Count,
            ct);
        
        product.MarkEventsAsCommitted();
        
        // 检查是否需要创建快照
        var lastSnapshotVersion = await _snapshotStore.GetLastSnapshotVersionAsync(product.Id, ct);
        
        if (_strategy.ShouldCreateSnapshot(product.Version, lastSnapshotVersion))
        {
            await CreateSnapshotAsync(product, ct);
        }
    }
    
    private async Task CreateSnapshotAsync(Product product, CancellationToken ct)
    {
        var snapshot = new AggregateSnapshot
        {
            AggregateId = product.Id,
            AggregateType = nameof(Product),
            Version = product.Version,
            SnapshotData = JsonSerializer.Serialize(new ProductSnapshot
            {
                Id = product.Id,
                Name = product.Name,
                Price = product.Price,
                Stock = product.Stock,
                Category = product.Category,
                IsActive = product.IsActive
            })
        };
        
        await _snapshotStore.SaveSnapshotAsync(snapshot, ct);
    }
    
    private Product DeserializeSnapshot(AggregateSnapshot snapshot)
    {
        var data = JsonSerializer.Deserialize<ProductSnapshot>(snapshot.SnapshotData);
        
        var product = new Product();
        // 直接从快照数据恢复状态,无需重放事件
        product.SetStateFromSnapshot(data);
        
        return product;
    }
}

// 投影: 从事件构建优化的读模型
public class ProductProjectionHandler
{
    private readonly ReadDbContext _readContext;
    
    public async Task HandleAsync(DomainEvent @event, CancellationToken ct = default)
    {
        switch (@event)
        {
            case ProductCreatedEvent e:
                await Handle(e, ct);
                break;
            case PriceChangedEvent e:
                await Handle(e, ct);
                break;
            case StockAdjustedEvent e:
                await Handle(e, ct);
                break;
        }
    }
    
    private async Task Handle(ProductCreatedEvent @event, CancellationToken ct)
    {
        _readContext.ProductReadModels.Add(new ProductReadModel
        {
            ProductId = @event.ProductId,
            Name = @event.Name,
            Price = @event.Price,
            Stock = @event.Stock,
            Category = @event.Category,
            IsActive = true,
            CreatedAt = @event.OccurredOn,
            LastUpdatedAt = @event.OccurredOn
        });
        
        await _readContext.SaveChangesAsync(ct);
    }
    
    private async Task Handle(PriceChangedEvent @event, CancellationToken ct)
    {
        var product = await _readContext.ProductReadModels
            .FirstOrDefaultAsync(p => p.ProductId == @event.ProductId, ct);
        
        if (product != null)
        {
            product.Price = @event.NewPrice;
            product.LastUpdatedAt = @event.OccurredOn;
            
            await _readContext.SaveChangesAsync(ct);
        }
    }
    
    private async Task Handle(StockAdjustedEvent @event, CancellationToken ct)
    {
        var product = await _readContext.ProductReadModels
            .FirstOrDefaultAsync(p => p.ProductId == @event.ProductId, ct);
        
        if (product != null)
        {
            product.Stock = @event.RemainingStock;
            product.LastUpdatedAt = @event.OccurredOn;
            
            await _readContext.SaveChangesAsync(ct);
        }
    }
}

csharp
public class ProjectionService : BackgroundService
{
    private readonly IServiceProvider _serviceProvider;
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await ProcessNewEventsAsync(stoppingToken);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Projection failed: {ex.Message}");
            }
            
            await Task.Delay(TimeSpan.FromSeconds(1), stoppingToken);
        }
    }
    
    private async Task ProcessNewEventsAsync(CancellationToken ct)
    {
        using var scope = _serviceProvider.CreateScope();
        var eventStore = scope.ServiceProvider.GetRequiredService<IEventStore>();
        var projectionHandler = scope.ServiceProvider.GetRequiredService<ProductProjectionHandler>();
        var checkpointStore = scope.ServiceProvider.GetRequiredService<ICheckpointStore>();
        
        // 获取上次处理的位置
        var lastProcessedPosition = await checkpointStore.GetLastCheckpointAsync(ct);
        
        // 获取新事件
        var newEvents = await eventStore.GetEventsAfterAsync(lastProcessedPosition, ct);
        
        foreach (var @event in newEvents)
        {
            await projectionHandler.HandleAsync(@event, ct);
            
            // 更新检查点
            await checkpointStore.UpdateCheckpointAsync(@event.EventId, ct);
        }
    }
}

7.1 重现历史状态 ​

csharp
public class TimeTravelService
{
    private readonly IEventStore _eventStore;
    
    /// <summary>
    /// 获取聚合在指定时间点的状态
    /// </summary>
    public async Task<Product> GetStateAtTimeAsync(Guid productId, DateTime pointInTime)
    {
        // 获取该时间点之前的所有事件
        var events = await _eventStore.GetEventsAsync(productId);
        
        var historicalEvents = events
            .Where(e => e.OccurredOn <= pointInTime)
            .ToList();
        
        if (!historicalEvents.Any())
            throw new NotFoundException($"No events found before {pointInTime}");
        
        // 重放事件到指定时间点
        var product = new Product();
        product.LoadFromHistory(historicalEvents);
        
        return product;
    }
    
    /// <summary>
    /// 获取聚合在指定版本的状态
    /// </summary>
    public async Task<Product> GetStateAtVersionAsync(Guid productId, int version)
    {
        var events = await _eventStore.GetEventsAsync(productId, 1, version);
        
        var product = new Product();
        product.LoadFromHistory(events);
        
        return product;
    }
    
    /// <summary>
    /// 比较两个版本之间的差异
    /// </summary>
    public async Task<StateComparison> CompareVersionsAsync(Guid productId, int version1, int version2)
    {
        var state1 = await GetStateAtVersionAsync(productId, version1);
        var state2 = await GetStateAtVersionAsync(productId, version2);
        
        return new StateComparison
        {
            Version1 = version1,
            Version2 = version2,
            Changes = new Dictionary<string, StateChange>
            {
                ["Name"] = new StateChange(state1.Name, state2.Name),
                ["Price"] = new StateChange(state1.Price, state2.Price),
                ["Stock"] = new StateChange(state1.Stock, state2.Stock)
            }
        };
    }
}

public class StateComparison
{
    public int Version1 { get; set; }
    public int Version2 { get; set; }
    public Dictionary<string, StateChange> Changes { get; set; } = new();
}

public record StateChange(object? OldValue, object? NewValue);
