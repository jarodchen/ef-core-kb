CQRS模式(Command Query Responsibility Segregation) ​

CQRS(命令查询职责分离)是一种架构模式,将读操作和写操作分离为不同的模型,从而优化性能、可扩展性和维护性。本章将深入探讨如何在 EF Core 应用中实现 CQRS 模式,包括命令处理、查询优化、事件驱动更新以及最终一致性管理。

1.1 什么是CQRS? ​

核心概念:

CQRS的优势:

✅ 性能优化 - 读取模型针对查询优化,写入模型针对业务逻辑
✅ 可扩展性 - 读写可以独立扩展(读多写少场景)
✅ 安全性 - 读取侧天然防注入,写入侧集中验证
✅ 灵活性 - 可以为不同场景定制不同的模型
✅ 可维护性 - 职责清晰,代码组织更好

适用场景:

✅ 推荐CQRS:

  • 读写比例悬殊(如100:1)
  • 复杂的业务逻辑和验证
  • 需要多种视图展示同一数据
  • 高并发读取场景
  • 团队协作开发

❌ 不推荐CQRS:

  • 简单的CRUD应用
  • 小型项目(增加复杂度)
  • 强一致性要求极高
  • 团队规模小且经验不足

1.2 CQRS vs CRUD对比 ​

csharp
// ===== 传统CRUD =====
public class ProductService_CRUD
{
    private readonly AppDbContext _context;
    
    // 读取 - 使用领域实体
    public async Task<Product> GetProductAsync(int id)
    {
        return await _context.Products.FindAsync(id);
        // 问题: 加载了所有字段,包括不需要的
    }
    
    // 写入 - 使用同一个领域实体
    public async Task UpdatePriceAsync(int id, decimal newPrice)
    {
        var product = await _context.Products.FindAsync(id);
        product.Price = newPrice;
        await _context.SaveChangesAsync();
        // 问题: 可能触发不必要的变更追踪
    }
}

// ===== CQRS =====
// 读取模型 - 轻量级DTO
public class ProductDto
{
    public int Id { get; set; }
    public string Name { get; set; }
    public decimal Price { get; set; }
    // 只包含需要的字段
}

// 写入模型 - 丰富的领域实体
public class Product : AggregateRoot
{
    public void UpdatePrice(decimal newPrice)
    {
        // 丰富的业务逻辑
        if (newPrice < 0) throw new ArgumentException("Price cannot be negative");
        if (newPrice > Cost * 10) throw new BusinessException("Price too high");
        
        Price = newPrice;
        AddDomainEvent(new ProductPriceChangedEvent(Id, Price, newPrice));
    }
}

// 查询服务 -  optimized for read
public class ProductQueryService
{
    public async Task<ProductDto> GetProductAsync(int id)
    {
        return await _context.Products
            .AsNoTracking()
            .Where(p => p.Id == id)
            .Select(p => new ProductDto 
            { 
                Id = p.Id, 
                Name = p.Name, 
                Price = p.Price 
            })
            .FirstOrDefaultAsync();
    }
}

// 命令服务 - optimized for write
public class ProductCommandService
{
    public async Task UpdatePriceAsync(int id, decimal newPrice)
    {
        var product = await _context.Products.FindAsync(id);
        product.UpdatePrice(newPrice);  // 领域方法
        await _context.SaveChangesAsync();
    }
}

3.2 命令处理器 ​

csharp
// 命令处理器接口
public interface ICommandHandler<in TCommand> where TCommand : ICommand
{
    Task HandleAsync(TCommand command, CancellationToken ct = default);
}

public interface ICommandHandler<in TCommand, TResult> where TCommand : ICommand<TResult>
{
    Task<TResult> HandleAsync(TCommand command, CancellationToken ct = default);
}

// 创建订单命令处理器
public class CreateOrderHandler : ICommandHandler<CreateOrderCommand, int>
{
    private readonly AppDbContext _context;
    private readonly IInventoryService _inventoryService;
    private readonly IPricingService _pricingService;
    private readonly ILogger<CreateOrderHandler> _logger;
    
    public CreateOrderHandler(
        AppDbContext context,
        IInventoryService inventoryService,
        IPricingService pricingService,
        ILogger<CreateOrderHandler> logger)
    {
        _context = context;
        _inventoryService = inventoryService;
        _pricingService = pricingService;
        _logger = logger;
    }
    
    public async Task<int> HandleAsync(CreateOrderCommand command, CancellationToken ct = default)
    {
        _logger.LogInformation("Processing order creation for customer {CustomerId}", 
            command.CustomerId);
        
        // 1. 验证库存
        foreach (var item in command.Items)
        {
            var hasStock = await _inventoryService.CheckStockAsync(
                item.ProductId, item.Quantity, ct);
            
            if (!hasStock)
                throw new InventoryException($"Product {item.ProductId} out of stock");
        }
        
        // 2. 创建订单项
        var orderItems = new List<OrderItem>();
        foreach (var itemCommand in command.Items)
        {
            // 验证价格
            var currentPrice = await _pricingService.GetCurrentPriceAsync(
                itemCommand.ProductId, ct);
            
            if (Math.Abs(itemCommand.UnitPrice - currentPrice) > 0.01m)
                throw new PricingException("Price mismatch");
            
            orderItems.Add(new OrderItem
            {
                ProductId = itemCommand.ProductId,
                Quantity = itemCommand.Quantity,
                UnitPrice = itemCommand.UnitPrice
            });
        }
        
        // 3. 创建订单(领域逻辑)
        var order = Order.Create(
            command.CustomerId,
            orderItems,
            command.ShippingAddress);
        
        _context.Orders.Add(order);
        
        // 4. 扣减库存
        foreach (var item in command.Items)
        {
            await _inventoryService.ReserveStockAsync(
                item.ProductId, item.Quantity, ct);
        }
        
        // 5. 保存
        await _context.SaveChangesAsync(ct);
        
        _logger.LogInformation("Order {OrderId} created successfully", order.Id);
        
        return order.Id;
    }
}

// FluentValidation验证器
public class CreateOrderCommandValidator : AbstractValidator<CreateOrderCommand>
{
    public CreateOrderCommandValidator()
    {
        RuleFor(x => x.CustomerId)
            .NotEmpty()
            .MaximumLength(100);
        
        RuleFor(x => x.Items)
            .NotEmpty()
            .Must(items => items.Count > 0 && items.Count <= 100)
            .WithMessage("Order must have 1-100 items");
        
        RuleForEach(x => x.Items).ChildRules(item =>
        {
            item.RuleFor(i => i.ProductId)
                .GreaterThan(0);
            
            item.RuleFor(i => i.Quantity)
                .InclusiveBetween(1, 999);
            
            item.RuleFor(i => i.UnitPrice)
                .GreaterThan(0)
                .LessThan(1000000);
        });
        
        RuleFor(x => x.ShippingAddress)
            .NotEmpty()
            .MaximumLength(500);
    }
}

// 验证中间件
public class ValidationMiddleware<TCommand> where TCommand : ICommand
{
    private readonly IEnumerable<IValidator<TCommand>> _validators;
    
    public async Task ValidateAsync(TCommand command)
    {
        var context = new ValidationContext<TCommand>(command);
        
        var failures = _validators
            .Select(v => v.Validate(context))
            .SelectMany(result => result.Errors)
            .Where(f => f != null)
            .ToList();
        
        if (failures.Any())
            throw new ValidationException(failures);
    }
}

app.MapPost("/api/orders", async (

    return Results.NoContent();
})
.RequireAuthorization("Admin");

4.2 查询处理器 ​

csharp
// 查询处理器接口
public interface IQueryHandler<in TQuery, TResult> where TQuery : IQuery<TResult>
{
    Task<TResult> HandleAsync(TQuery query, CancellationToken ct = default);
}

// 获取订单查询处理器
public class GetOrderHandler : IQueryHandler<GetOrderQuery, OrderDetailDto>
{
    private readonly AppDbContext _context;
    
    public async Task<OrderDetailDto> HandleAsync(GetOrderQuery query, CancellationToken ct = default)
    {
        // 优化的查询 - 只选择需要的字段
        var order = await _context.Orders
            .AsNoTracking()  // 重要: 禁用变更追踪
            .Where(o => o.Id == query.OrderId)
            .Select(o => new OrderDetailDto
            {
                Id = o.Id,
                OrderDate = o.OrderDate,
                CustomerId = o.CustomerId,
                CustomerName = o.Customer.Name,  // JOIN
                Status = o.Status.ToString(),
                TotalAmount = o.TotalAmount,
                ShippingAddress = o.ShippingAddress,
                Items = o.Items.Select(i => new OrderItemDto
                {
                    ProductId = i.ProductId,
                    ProductName = i.Product.Name,
                    Quantity = i.Quantity,
                    UnitPrice = i.UnitPrice,
                    Subtotal = i.Quantity * i.UnitPrice
                }).ToList()
            })
            .FirstOrDefaultAsync(ct);
        
        if (order == null)
            throw new NotFoundException($"Order {query.OrderId} not found");
        
        return order;
    }
}

// 产品列表查询处理器
public class ListProductsHandler : IQueryHandler<ListProductsQuery, PagedResult<ProductDto>>
{
    private readonly AppDbContext _context;
    
    public async Task<PagedResult<ProductDto>> HandleAsync(
        ListProductsQuery query, CancellationToken ct = default)
    {
        var qry = _context.Products.AsNoTracking();
        
        // 动态过滤
        if (query.CategoryId.HasValue)
            qry = qry.Where(p => p.CategoryId == query.CategoryId);
        
        if (query.MinPrice.HasValue)
            qry = qry.Where(p => p.Price >= query.MinPrice);
        
        if (query.MaxPrice.HasValue)
            qry = qry.Where(p => p.Price <= query.MaxPrice);
        
        if (!string.IsNullOrEmpty(query.SearchTerm))
            qry = qry.Where(p => p.Name.Contains(query.SearchTerm));
        
        // 总数
        var totalCount = await qry.CountAsync(ct);
        
        // 排序
        qry = query.SortBy.ToLower() switch
        {
            "name" => query.SortDescending 
                ? qry.OrderByDescending(p => p.Name)
                : qry.OrderBy(p => p.Name),
            "price" => query.SortDescending 
                ? qry.OrderByDescending(p => p.Price)
                : qry.OrderBy(p => p.Price),
            _ => query.SortDescending 
                ? qry.OrderByDescending(p => p.CreatedAt)
                : qry.OrderBy(p => p.CreatedAt)
        };
        
        // 分页
        var items = await qry
            .Skip((query.Page - 1) * query.PageSize)
            .Take(query.PageSize)
            .Select(p => new ProductDto
            {
                Id = p.Id,
                Name = p.Name,
                Price = p.Price,
                Stock = p.Stock,
                CategoryName = p.Category.Name
            })
            .ToListAsync(ct);
        
        return new PagedResult<ProductDto>
        {
            Items = items,
            TotalCount = totalCount,
            Page = query.Page,
            PageSize = query.PageSize
        };
    }
}

-- 创建物化视图
CREATE VIEW v_ProductSummary AS
SELECT 
    p.Id,
    p.Name,
    p.Price,
    p.Stock,
    c.Name AS CategoryName,
    COUNT(DISTINCT oi.OrderId) AS TotalOrders,
    SUM(oi.Quantity) AS TotalQuantitySold,
    AVG(r.Rating) AS AvgRating,
    COUNT(r.Id) AS ReviewCount
FROM Products p
LEFT JOIN Categories c ON p.CategoryId = c.Id
LEFT JOIN OrderItems oi ON p.Id = oi.ProductId
LEFT JOIN Reviews r ON p.Id = r.ProductId
GROUP BY p.Id, p.Name, p.Price, p.Stock, c.Name;

-- 定期刷新
CREATE PROCEDURE sp_RefreshProductSummary
AS
BEGIN
    -- 更新物化视图
    UPDATE ps
    SET 
        ps.TotalOrders = stats.TotalOrders,
        ps.TotalQuantitySold = stats.TotalQuantitySold,
        ps.AvgRating = stats.AvgRating
    FROM v_ProductSummary ps
    INNER JOIN (
        SELECT 
            ProductId,
            COUNT(DISTINCT OrderId) AS TotalOrders,
            SUM(Quantity) AS TotalQuantitySold
        FROM OrderItems
        GROUP BY ProductId
    ) stats ON ps.Id = stats.ProductId;
END

public class ProductSummaryHandler : IQueryHandler<ListProductsQuery, PagedResult<ProductSummaryDto>>
{
    public async Task<PagedResult<ProductSummaryDto>> HandleAsync(
        ListProductsQuery query, CancellationToken ct = default)
    {
        // 超快查询 - 从物化视图读取
        var items = await _context.ProductSummaries
            .AsNoTracking()
            .Where(ps => !query.CategoryId.HasValue || ps.CategoryName == query.CategoryId.ToString())
            .OrderByDescending(ps => ps.TotalOrders)
            .Skip((query.Page - 1) * query.PageSize)
            .Take(query.PageSize)
            .ToListAsync(ct);
        
        var totalCount = await _context.ProductSummaries.CountAsync(ct);
        
        return new PagedResult<ProductSummaryDto>
        {
            Items = items,
            TotalCount = totalCount,
            Page = query.Page,
            PageSize = query.PageSize
        };
    }
}

csharp
public class ReadModelSyncService : BackgroundService
{
    private readonly IServiceProvider _serviceProvider;
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await SyncReadModelsAsync(stoppingToken);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Sync failed: {ex.Message}");
            }
            
            // 每5秒同步一次
            await Task.Delay(TimeSpan.FromSeconds(5), stoppingToken);
        }
    }
    
    private async Task SyncReadModelsAsync(CancellationToken ct)
    {
        using var scope = _serviceProvider.CreateScope();
        var writeContext = scope.ServiceProvider.GetRequiredService<WriteDbContext>();
        var readContext = scope.ServiceProvider.GetRequiredService<ReadDbContext>();
        
        // 同步产品统计
        var products = await writeContext.Products
            .Include(p => p.Category)
            .ToListAsync(ct);
        
        foreach (var product in products)
        {
            var summary = await readContext.ProductSummaries
                .FirstOrDefaultAsync(ps => ps.ProductId == product.Id, ct);
            
            if (summary == null)
            {
                readContext.ProductSummaries.Add(new ProductSummary
                {
                    ProductId = product.Id,
                    Name = product.Name,
                    Price = product.Price,
                    CategoryName = product.Category?.Name
                });
            }
            else
            {
                summary.Price = product.Price;
                summary.Name = product.Name;
            }
        }
        
        await readContext.SaveChangesAsync(ct);
    }
}

public class ConsistencyMonitor
{
    private readonly WriteDbContext _writeContext;
    private readonly ReadDbContext _readContext;
    private readonly ILogger<ConsistencyMonitor> _logger;
    
    public async Task CheckConsistencyAsync()
    {
        // 检查写数据库和读数据库的一致性
        var writeCount = await _writeContext.Orders.CountAsync();
        var readCount = await _readContext.OrderSummaries.CountAsync();
        
        if (writeCount != readCount)
        {
            _logger.LogWarning("Inconsistency detected: Write={WriteCount}, Read={ReadCount}",
                writeCount, readCount);
            
            // 触发重新同步
            await TriggerResyncAsync();
        }
    }
}
