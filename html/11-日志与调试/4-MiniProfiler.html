MiniProfiler - 性能分析工具 ​

概述 ​

MiniProfiler 是一个轻量级的性能分析工具,可以实时监控 EF Core 查询性能,包括 SQL 执行时间、查询次数、N+1 问题检测等。它提供了直观的 Web UI,帮助开发者快速定位性能瓶颈。

核心功能 ​

功能说明
SQL 监控记录每个 SQL 查询的执行时间和参数
N+1 检测自动识别 N+1 查询问题
调用堆栈显示查询的 C# 代码位置
自定义步骤标记业务逻辑步骤的性能
历史对比比较不同请求的性能差异
生产环境支持可选择性在生产环境启用

基础配置 ​

csharp
// Program.cs
using StackExchange.Profiling;

var builder = WebApplication.CreateBuilder(args);

// 添加 MiniProfiler 服务
builder.Services.AddMiniProfiler(options =>
{
    // 路由前缀(访问 /profiler 查看结果)
    options.RouteBasePath = "/profiler";
    
    // UI 配置
    options.ColorScheme = ColorScheme.Dark;  // 深色主题
    options.PopupRenderPosition = RenderPosition.Right;  // 浮窗位置
    options.ShowControls = true;  // 显示控制按钮
    
    // 授权配置(生产环境重要!)
    options.ResultsAuthorize = request => 
        request.HttpContext.User.IsInRole("Admin");
    options.ResultsListAuthorize = options.ResultsAuthorize;
    
    // 采样率(生产环境降低采样率)
    options.SampleRate = 100;  // 100% 采样(开发环境)
    // options.SampleRate = 1;  // 1% 采样(生产环境)
})
.AddEntityFramework();  // ← 集成 EF Core

var app = builder.Build();

// 使用中间件
if (app.Environment.IsDevelopment())
{
    app.UseMiniProfiler();  // 仅开发环境启用
}

app.Run();

自定义监控步骤 ​

csharp
app.MapGet("/api/orders/{id}/details", async (int id, AppDbContext db) =>
{
    // 自定义步骤 1: 加载订单
    Order? order;
    using (MiniProfiler.Current.Step("Load Order"))
    {
        order = await db.Orders
            .Include(o => o.Items)
            .FirstOrDefaultAsync(o => o.Id == id);
    }
    
    if (order == null)
        return Results.NotFound();
    
    // 自定义步骤 2: 加载客户信息
    Customer? customer;
    using (MiniProfiler.Current.Step("Load Customer"))
    {
        customer = await db.Customers
            .FirstOrDefaultAsync(c => c.Id == order.CustomerId);
    }
    
    // 自定义步骤 3: 计算统计数据
    OrderStatistics stats;
    using (MiniProfiler.Current.Step("Calculate Statistics"))
    {
        stats = await CalculateOrderStatistics(db, order.Id);
    }
    
    return Results.Ok(new 
    { 
        Order = order, 
        Customer = customer, 
        Statistics = stats 
    });
});

// MiniProfiler UI 会显示每个步骤的耗时

标记关键代码块 ​

csharp
public class ProductService
{
    private readonly AppDbContext _context;

    public async Task<List<Product>> SearchProductsAsync(string searchTerm)
    {
        // 标记整个方法
        using var profiler = MiniProfiler.Current.Step(nameof(SearchProductsAsync));
        
        // 子步骤 1: 验证输入
        using (MiniProfiler.Current.Step("Validate Input"))
        {
            if (string.IsNullOrWhiteSpace(searchTerm))
                throw new ArgumentException("Search term is required");
        }
        
        // 子步骤 2: 执行搜索
        List<Product> products;
        using (MiniProfiler.Current.Step("Execute Search Query"))
        {
            products = await _context.Products
                .Where(p => p.Name.Contains(searchTerm) || 
                           p.Description.Contains(searchTerm))
                .ToListAsync();
        }
        
        // 子步骤 3: 后处理
        using (MiniProfiler.Current.Step("Post-process Results"))
        {
            // 格式化、过滤等
            products = products.Take(100).ToList();
        }
        
        return products;
    }
}

实际案例分析 ​

案例 1: 识别慢查询 ​

csharp
// 问题: API 响应缓慢
app.MapGet("/api/reports/sales", async (AppDbContext db) =>
{
    // MiniProfiler 显示此查询耗时 2.5 秒!
    var sales = await db.Orders
        .Where(o => o.OrderDate >= DateTime.UtcNow.AddMonths(-12))
        .Select(o => new 
        { 
            o.OrderDate,
            o.TotalAmount,
            Customer = o.Customer.Name,  // ⚠️ 隐式 JOIN
            Items = o.Items.Select(i => new 
            { 
                i.ProductName, 
                i.Quantity,
                i.UnitPrice 
            })  // ⚠️ N+1 问题
        })
        .ToListAsync();
    
    return Results.Ok(sales);
});

// MiniProfiler 输出:
/*
⚠️ Slow Query Detected (2,543ms)

SELECT [o].[OrderDate], [o].[TotalAmount], [c].[Name], ...
FROM [Orders] AS [o]
INNER JOIN [Customers] AS [c] ON [o].[CustomerId] = [c].[Id]
WHERE [o].[OrderDate] >= @__date_0

💡 Optimization Suggestions:
1. Add index on Orders.OrderDate
2. Consider pagination (currently loading all 12 months)
3. Project only needed columns
*/

// ✅ 优化后
app.MapGet("/api/reports/sales", async (
    AppDbContext db, 
    int page = 1, 
    int pageSize = 50) =>
{
    var sales = await db.Orders
        .Where(o => o.OrderDate >= DateTime.UtcNow.AddMonths(-3))  // 仅 3 个月
        .OrderByDescending(o => o.OrderDate)
        .Skip((page - 1) * pageSize)
        .Take(pageSize)
        .Select(o => new 
        { 
            o.OrderDate,
            o.TotalAmount,
            CustomerName = o.Customer.Name
        })
        .ToListAsync();
    
    return Results.Ok(sales);
});

// 优化后: 45ms (提升 98%)

案例 2: 检测 N+1 问题 ​

csharp
// 问题: 页面加载缓慢
app.MapGet("/api/blog/posts", async (AppDbContext db) =>
{
    var posts = await db.Posts.ToListAsync();
    
    foreach (var post in posts)
    {
        // ⚠️ N+1: 每个帖子都查询作者
        post.Author = await db.Users.FindAsync(post.AuthorId);
        
        // ⚠️ N+1: 每个帖子都查询评论
        post.Comments = await db.Comments
            .Where(c => c.PostId == post.Id)
            .ToListAsync();
    }
    
    return Results.Ok(posts);
});

// MiniProfiler 警告:
/*
⚠️ Performance Issues Detected

N+1 Query Pattern:
- Query 1: SELECT * FROM Posts (1 query, 5ms)
- Query 2: SELECT * FROM Users WHERE Id = @id (100 queries, 250ms) ⚠️
- Query 3: SELECT * FROM Comments WHERE PostId = @id (100 queries, 180ms) ⚠️

Total: 201 queries, 435ms

💡 Fix: Use Include() to eager load related data
*/

// ✅ 修复后
app.MapGet("/api/blog/posts", async (AppDbContext db) =>
{
    var posts = await db.Posts
        .Include(p => p.Author)
        .Include(p => p.Comments)
        .ToListAsync();
    
    return Results.Ok(posts);
});

// 优化后: 3 queries, 25ms (提升 94%)

生产环境部署 ​

安全配置 ​

csharp
if (!app.Environment.IsDevelopment())
{
    // 生产环境配置
    builder.Services.AddMiniProfiler(options =>
    {
        options.RouteBasePath = "/profiler";
        
        // ⚠️ 严格授权: 仅管理员可访问
        options.ResultsAuthorize = request =>
        {
            var user = request.HttpContext.User;
            return user.Identity?.IsAuthenticated == true &&
                   user.IsInRole("Administrator") &&
                   IsInternalIp(request.HttpContext.Connection.RemoteIpAddress);
        };
        
        // 低采样率
        options.SampleRate = 1;  // 1% 采样
        
        // 限制存储数量
        options.MaxUnviewedProfiles = 10;
        options.StorageDuration = TimeSpan.FromHours(1);
        
        // 隐藏敏感信息
        options.SqlFormatter = new StackExchange.Profiling.SqlFormatters.InlineFormatter();
    })
    .AddEntityFramework();
    
    app.UseMiniProfiler();
}

bool IsInternalIp(IPAddress? ip)
{
    if (ip == null) return false;
    
    // 检查是否为内网 IP
    return ip.IsIPv6SiteLocal || 
           ip.ToString().StartsWith("10.") ||
           ip.ToString().StartsWith("192.168.") ||
           ip.ToString().StartsWith("172.16.");
}
