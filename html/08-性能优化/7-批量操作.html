csharp
// ❌ 传统方式: N+1 次数据库往返

    .Where(p => p.CategoryId == 1)

{
    product.Price *= 1.1m; // 涨价 10%

await context.SaveChangesAsync();

// 问题:

public class BulkOperationBenchmark
{
    private readonly AppDbContext _context;

    // 传统方式: 批量更新
    [Benchmark]
    public async Task TraditionalUpdate()
    {
        var products = await _context.Products
            .Where(p => p.CategoryId == 1)
            .ToListAsync();
        
        foreach (var product in products)
        {
            product.Price *= 1.1m;
        }
        await _context.SaveChangesAsync();
    }

    // 批量操作: ExecuteUpdate
    [Benchmark]
    public async Task BulkUpdate()
    {

            .Where(p => p.CategoryId == 1)

    {
        var products = await _context.Products
            .Where(p => !p.IsActive)
            .ToListAsync();
        
        _context.Products.RemoveRange(products);
        await _context.SaveChangesAsync();
    }

    // 批量操作: ExecuteDelete
    [Benchmark]
    public async Task BulkDelete()
    {
        await _context.Products
            .Where(p => !p.IsActive)
            .ExecuteDeleteAsync();
    }
}

❌ 不适合批量操作的场景 ​

1. 需要触发业务逻辑 ​

csharp
// ❌ 错误: 跳过业务验证
await context.Products
    .Where(p => p.Price < 0)
    .ExecuteDeleteAsync();

// ✅ 正确: 需要验证和清理关联数据
var products = await context.Products
    .Include(p => p.OrderItems)
    .Include(p => p.Reviews)
    .Where(p => p.Price < 0)
    .ToListAsync();

foreach (var product in products)
{
    // 业务验证
    if (product.OrderItems.Any())
    {
        throw new InvalidOperationException("Cannot delete product with orders");
    }
    
    // 清理关联数据
    context.Reviews.RemoveRange(product.Reviews);
    context.Products.Remove(product);
}

await context.SaveChangesAsync();

注意事项 ​

1. 不触发变更跟踪 ​

csharp
// ExecuteUpdate/Delete 不会触发 ChangeTracker
await context.Products
    .Where(p => p.CategoryId == 1)
    .ExecuteUpdateAsync(setters => setters
        .SetProperty(p => p.Price, 99.99m));

// ⚠️ 已加载的实体不会自动更新
var product = await context.Products.FindAsync(1);
Console.WriteLine(product.Price); // 旧值!

// ✅ 解决: 重新查询或清除缓存
context.ChangeTracker.Clear();
var updatedProduct = await context.Products.FindAsync(1);
Console.WriteLine(updatedProduct.Price); // 新值

4. 事务支持 ​

csharp
// ExecuteUpdate/Delete 支持事务
await using var transaction = await context.Database.BeginTransactionAsync();

try
{
    // 批量更新
    await context.Products
        .Where(p => p.CategoryId == 1)
        .ExecuteUpdateAsync(setters => setters
            .SetProperty(p => p.Price, p => p.Price * 1.1m));
    
    // 其他操作
    await context.Categories
        .Where(c => c.Id == 1)
        .ExecuteUpdateAsync(setters => setters
            .SetProperty(c => c.LastUpdated, DateTime.UtcNow));
    
    await transaction.CommitAsync();
}
catch (Exception)
{
    await transaction.RollbackAsync();
    throw;
}

3. 配合事务使用 ​

csharp
public async Task UpdatePricesAsync(int categoryId, decimal multiplier)
{
    await using var transaction = await context.Database.BeginTransactionAsync();
    
    try
    {
        // 批量更新价格
        await context.Products
            .Where(p => p.CategoryId == categoryId)
            .ExecuteUpdateAsync(setters => setters
                .SetProperty(p => p.Price, p => p.Price * multiplier));
        
        // 记录变更日志
        await context.PriceChangeLogs.AddAsync(new PriceChangeLog
        {
            CategoryId = categoryId,
            Multiplier = multiplier,
            ChangedAt = DateTime.UtcNow
        });
        
        await transaction.CommitAsync();
    }
    catch (Exception)
    {
        await transaction.RollbackAsync();
        throw;
    }
}

4. 监控批量操作 ​

csharp
public class BulkOperationMonitor
{
    private readonly ILogger<BulkOperationMonitor> _logger;

    public async Task MonitorBulkUpdateAsync(IQueryable<Product> query)
    {
        var stopwatch = Stopwatch.StartNew();
        
        var affectedRows = await query.ExecuteUpdateAsync(setters => setters
            .SetProperty(p => p.ModifiedAt, DateTime.UtcNow));
        
        stopwatch.Stop();
        
        _logger.LogInformation(
            "Bulk update completed: {Rows} rows in {Elapsed}ms",
            affectedRows,
            stopwatch.ElapsedMilliseconds);
    }
}

📊 性能优化检查清单 ​

使用批量操作前确认:

  • [ ] 是否需要触发业务逻辑?
  • [ ] 是否需要审计日志?
  • [ ] 是否需要并发控制?
  • [ ] 数据量是否大于 100 条?
  • [ ] 是否可以接受不触发验证?
  • [ ] 是否需要事务支持?

如果以上都为"否",使用批量操作!

💡 小结 ​

核心要点:

  • ✅ ExecuteUpdateAsync 批量更新（EF Core 8+）
  • ✅ ExecuteDeleteAsync 批量删除（EF Core 8+）
  • ✅ 性能提升 10-100 倍
  • ✅ 不加载实体到内存
  • ✅ 单次数据库往返
  • ❌ 不触发变更跟踪、验证、拦截器

使用原则:

简单批量操作 → ExecuteUpdate/Delete
需要业务逻辑 → 传统方式
EF Core 6/7 → EFCore.BulkExtensions
大批量数据 → 分批处理 + 事务
